19      @Test
20      void testFetchPosts_UsingMockito_NoHttpCall() throws Exception {
21          String json = """
22          [
23            {"userId":1,"id":101},
24            {"userId":2,"id":201}
25          ]
26      """;
27  
28          HttpClient mockClient = mock(HttpClient.class);
29  
30          @SuppressWarnings("unchecked")
31          HttpResponse<String> mockResponse = (HttpResponse<String>) mock(HttpResponse.class);
32  
33          when(mockResponse.body()).thenReturn(json);
34  
35          // Fix: Use doReturn to avoid Java compiler generic mismatch on send()
36          doReturn(mockResponse).when(mockClient).send(any(HttpRequest.class), any());
37  
38          try (MockedStatic<HttpClient> mockedStatic = mockStatic(HttpClient.class)) {
39              mockedStatic.when(HttpClient::newHttpClient).thenReturn(mockClient);
40  
41              var posts = PostAnalyticsClient.fetchPosts();
42  
43              assertNotNull(posts);
44              assertEquals(2, posts.size());
45  
46              mockedStatic.verify(HttpClient::newHttpClient);
47              verify(mockClient, times(1)).send(any(HttpRequest.class), any());
48          }
49      }
