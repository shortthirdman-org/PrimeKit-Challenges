11  
12  class SensorDigitAnalyzerTest {
13  
14      SensorDigitAnalyzer app;
15  
16      @BeforeEach
17      void setUp() {
18          app =  new SensorDigitAnalyzer();
19      }
20  
21      // Positive Test: Basic example (classic and stream)
22      @Test
23      void testBasicExample() {
24          int[] readingsArray = {123, 456, 789, 101};
25          List<Integer> readingsList = Arrays.asList(123, 456, 789, 101);
26  
27          assertEquals(6, app.findMostFrequentSingleDigit(readingsArray));
28          assertEquals(6, app.findMostFrequentSingleDigit(readingsList));
29      }
30  
31      // Positive Test: Tie-breaking case (highest digit wins)
32      @Test
33      void testTieBreaker() {
34          int[] readingsArray = {19, 28, 37, 46}; // Reduces to [1,1,1,1] => Only '1' appears
35          List<Integer> readingsList = Arrays.asList(9, 18, 27, 36);    // Reduces to [9,9,9,9] => Only '9' appears
36  
37          assertEquals(1, app.findMostFrequentSingleDigit(readingsArray));
38          assertEquals(9, app.findMostFrequentSingleDigit(readingsList));
39      }
40  
41      // Positive Test: Mixed digits with tie (choose higher)
42      @Test
43      void testTieWithHigherDigitWinning() {
44          int[] readingsArray = {10, 100, 55, 46}; // Reduces to [1,1,1,5] => 1(3x), 5(1x)
45          List<Integer> readingsList = Arrays.asList(11, 20, 2, 29);    // Reduces to [2,2,2,2] => Only 2
46  
47          assertEquals(1, app.findMostFrequentSingleDigit(readingsArray));
48          assertEquals(2, app.findMostFrequentSingleDigit(readingsList));
49      }
50  
51      // Edge Test: Empty array/list
52      @Test
53      void testEmptyInput() {
54          int[] emptyArray = {};
55          List<Integer> emptyList = Collections.emptyList();
56  
57          assertEquals(-1, app.findMostFrequentSingleDigit(emptyArray));
58          assertEquals(-1, app.findMostFrequentSingleDigit(emptyList));
59      }
60  
61      // Edge Test: All elements reduce to same digit
62      @Test
63      void testAllSameReducedDigit() {
64          int[] readingsArray = {111, 12, 21, 3};   // All reduce to 3
65          List<Integer> readingsList = Arrays.asList(222, 6, 33); // All reduce to 6
66  
67          assertEquals(3, app.findMostFrequentSingleDigit(readingsArray));
68          assertEquals(6, app.findMostFrequentSingleDigit(readingsList));
69      }
70  
71      // Edge Test: Single element array/list
72      @Test
73      void testSingleElement() {
74          int[] singleArray = {9999}; // Reduces to 9
75          List<Integer> singleList = Collections.singletonList(123456); // Reduces to 3
76  
77          assertEquals(9, app.findMostFrequentSingleDigit(singleArray));
78          assertEquals(3, app.findMostFrequentSingleDigit(singleList));
79      }
80  
81      // Edge Test: All zeroes
82      @Test
83      void testAllZeroes() {
84          int[] zeroArray = {0, 0, 0};
85          List<Integer> zeroList = Arrays.asList(0, 0, 0);
86  
87          assertEquals(0, app.findMostFrequentSingleDigit(zeroArray));
88          assertEquals(0, app.findMostFrequentSingleDigit(zeroList));
89      }
90  
91      // Edge Test: Large values
92      @Test
93      void testLargeValues() {
94          int[] largeArray = {999999999, 888888888, 777777777}; // All reduce to 9
95          List<Integer> largeList = Arrays.asList(1000000000, 1234567890); // Reduce to 1 and 9 -> 9 wins
96  
97          assertEquals(9, app.findMostFrequentSingleDigit(largeArray));
98          assertEquals(9, app.findMostFrequentSingleDigit(largeList));
99      }
