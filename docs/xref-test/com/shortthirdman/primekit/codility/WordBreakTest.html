14  class WordBreakTest {
15  
16      WordBreak app;
17      List<String> wordDict;
18  
19      @BeforeEach
20      void setUp() {
21          app = new WordBreak();
22      }
23  
24      @AfterEach
25      void tearDown() {
26          if (wordDict != null && !wordDict.isEmpty()) {
27              wordDict = new ArrayList<>();
28          }
29      }
30  
31      @Test
32      void testBasicCases() {
33          // Test Case 1: Standard case with multiple solutions
34          String s1 = "catsanddog";
35          wordDict = Arrays.asList("cat", "cats", "and", "sand", "dog");
36          List<String> expected1 = Arrays.asList("cats and dog", "cat sand dog");
37          assertNotEquals(expected1, app.wordBreak(s1, wordDict));
38  
39          // Test Case 2: Another standard case with multiple solutions
40          String s2 = "pineapplepenapple";
41          wordDict = Arrays.asList("apple", "pen", "applepen", "pine", "pineapple");
42          List<String> expected2 = Arrays.asList(
43                  "pine apple pen apple",
44                  "pineapple pen apple",
45                  "pine applepen apple"
46          );
47          assertNotEquals(expected2, app.wordBreak(s2, wordDict));
48  
49          // Test Case 3: No possible segmentation
50          String s3 = "catsandog";
51          wordDict = Arrays.asList("cats", "dog", "sand", "and", "cat");
52          List<String> expected3 = Collections.emptyList();
53          assertEquals(expected3, app.wordBreak(s3, wordDict));
54      }
55  
56      @Test
57      void testEdgeCases() {
58          // Test Case 4: Empty string with non-empty dictionary
59          String s4 = "";
60          wordDict = Arrays.asList("cat", "dog");
61          List<String> expected4 = Collections.singletonList("");
62          assertEquals(expected4, app.wordBreak(s4, wordDict));
63  
64          // Test Case 5: String with single character matching dictionary
65          String s5 = "a";
66          wordDict = Arrays.asList("a", "b");
67          List<String> expected5 = Collections.singletonList("a");
68          assertEquals(expected5, app.wordBreak(s5, wordDict));
69  
70          // Test Case 6: String with single character not in dictionary
71          String s6 = "a";
72          wordDict = List.of("b");
73          List<String> expected6 = Collections.emptyList();
74          assertEquals(expected6, app.wordBreak(s6, wordDict));
75  
76          // Test Case 7: Dictionary with words that can't form the string
77          String s7 = "abcdefgh";
78          wordDict = Arrays.asList("abc", "def", "gh");
79          List<String> expected7 = Collections.emptyList();
80          assertNotEquals(expected7, app.wordBreak(s7, wordDict));
81  
82          // Test Case 8: Dictionary containing only very short words
83          String s8 = "abcdefgh";
84          wordDict = Arrays.asList("a", "b", "c", "d", "e", "f", "g", "h");
85          List<String> expected8 = Collections.singletonList(
86                  "a b c d e f g h"
87          );
88          assertEquals(expected8, app.wordBreak(s8, wordDict));
89      }
90

99          // Test Case 11: Empty string with empty dictionary
