10  
11      JumpGame game;

20      }

22      @Test
23      void jump() {
24          // ✅ Provided examples
25          assertEquals(2, game.jump(new int[]{2, 3, 1, 1, 4}),
26                  "Should need 2 jumps");
27          assertEquals(2, game.jump(new int[]{2, 3, 0, 1, 4}),
28                  "Should need 2 jumps");
29  
30          // ✅ Edge cases
31          assertEquals(0, game.jump(new int[]{0}),
32                  "Single element means no jumps needed");
33          assertEquals(1, game.jump(new int[]{1, 0}),
34                  "Direct jump to last index");
35          assertEquals(1, game.jump(new int[]{5, 0, 0, 0, 0}),
36                  "Big jump covers everything in one go");
37          assertEquals(3, game.jump(new int[]{1, 1, 1, 1}),
38                  "Must jump each step to move forward");
39  
40          // ❌ Constraint validation
41          assertThrows(IllegalArgumentException.class,
42                  () -> game.jump(null),
43                  "Null input should throw exception");
44          assertThrows(IllegalArgumentException.class,
45                  () -> game.jump(new int[0]),
46                  "Empty array should throw exception");
47          assertThrows(IllegalArgumentException.class,
48                  () -> game.jump(new int[10_001]),
49                  "Array longer than 10^4 should throw exception");
50          assertThrows(IllegalArgumentException.class,
51                  () -> game.jump(new int[]{-1, 2, 3}),
52                  "Negative values should throw exception");
53          assertThrows(IllegalArgumentException.class,
54                  () -> game.jump(new int[]{1001}),
55                  "Values greater than 1000 should throw exception");
56  
57          // 🛠 More edge validations
58          assertEquals(2, game.jump(new int[]{3, 2, 1, 0, 4}),
59                  "Greedy should still find the optimal 2 jumps");
60          assertEquals(2, game.jump(new int[]{4, 1, 1, 3, 1, 1, 1}),
61                  "First jump to index 3, second jump to last");
62      }
63  
64      @Test
65      void canJump() {
66          // ✅ Positive cases
67          assertTrue(game.canJump(new int[]{2, 3, 1, 1, 4}),
68                  "Should be able to reach the last index");
69          assertTrue(game.canJump(new int[]{1}),
70                  "Single element array should always return true");
71          assertTrue(game.canJump(new int[]{5, 0, 0, 0, 0}),
72                  "Large jump at the start should allow reaching the end");
73  
74          // ❌ Negative cases
75          assertFalse(game.canJump(new int[]{3, 2, 1, 0, 4}),
76                  "Should not be able to reach the last index");
77          assertFalse(game.canJump(new int[]{0, 2, 3}),
78                  "Cannot move past index 0 with zero jump length");
79          assertFalse(game.canJump(new int[]{1, 0, 1, 0}),
80                  "Should get stuck at the second zero");
81  
82          // 🛠 Edge cases
83          assertTrue(game.canJump(new int[]{0}),
84                  "Array with single zero is trivially at the end");
85          assertTrue(game.canJump(new int[]{2, 5, 0, 0}),
86                  "Should be able to leap over trailing zeros");
87          assertFalse(game.canJump(new int[]{1, 1, 0, 0}),
88                  "Should get stuck before last index");
89          assertTrue(game.canJump(new int[]{9, 0, 0, 0, 0, 0, 0, 0, 0, 0}),
90                  "Very large jump should instantly reach the end");
91          assertFalse(game.canJump(new int[]{1, 2, 1, 0, 0, 0}),
92                  "Trailing zeros without enough reach should fail");
93          assertTrue(game.canJump(new int[]{1, 2, 3, 4, 0, 0, 1}),
94                  "Last element itself is reachable despite intermediate zeros");
95  
96          // 🛑 Constraint validation
97          assertThrows(IllegalArgumentException.class,
98                  () -> game.canJump(null),
99                  "Null array should throw exception");
100         assertThrows(IllegalArgumentException.class,
101                 () -> game.canJump(new int[0]),
102                 "Empty array should throw exception");
103         assertThrows(IllegalArgumentException.class,
104                 () -> game.canJump(new int[10_001]),
105                 "Array longer than 10^4 should throw exception");
106         assertThrows(IllegalArgumentException.class,
107                 () -> game.canJump(new int[]{-1, 2, 3}),
108                 "Negative values should throw exception");
109         assertThrows(IllegalArgumentException.class,
110                 () -> game.canJump(new int[]{100_001}),
111                 "Values greater than 10^5 should throw exception");
112     }
