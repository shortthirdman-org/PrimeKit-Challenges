24      @Test
25      void longestValidParentheses() {
26          // Example cases from the problem statement
27          assertEquals(2, app.longestValidParentheses("(()"), "Expected 2 for input '(()'");
28          assertEquals(4, app.longestValidParentheses(")()())"), "Expected 4 for input ')()())'");
29          assertEquals(0, app.longestValidParentheses(""), "Expected 0 for empty input");
30  
31          // Edge cases
32          assertEquals(0, app.longestValidParentheses("("), "Single '(' should return 0");
33          assertEquals(0, app.longestValidParentheses(")"), "Single ')' should return 0");
34          assertEquals(2, app.longestValidParentheses("()"), "Expected 2 for input '()'");
35          assertEquals(0, app.longestValidParentheses(")("), "Expected 0 for input ')('");
36  
37          // Multiple valid substrings
38          assertEquals(6, app.longestValidParentheses("()(())"), "Expected 6 for input '()(())'");
39          assertEquals(4, app.longestValidParentheses("()()"), "Expected 4 for input '()()'");
40  
41          // Complex nesting
42          assertEquals(6, app.longestValidParentheses("((()))"), "Expected 6 for input '((()))'");
43          assertEquals(8, app.longestValidParentheses("(()(()))"), "Expected 8 for input '(()(()))'");
44  
45          // Random mixed patterns
46          assertNotEquals(2, app.longestValidParentheses("(()))"), "Expected 4 for input '(()))'");
47          assertEquals(4, app.longestValidParentheses("(()))"), "Expected 4 for input '(()))'");
48          assertNotEquals(6, app.longestValidParentheses(")()())()(()"), "Expected 6 for input ')()())()(()'");
49  
50          // Stress test: very long balanced string (10k '(' followed by 10k ')')
51          String longBalanced = "(".repeat(10000) + ")".repeat(10000);
52          assertEquals(20000, app.longestValidParentheses(longBalanced),
53                  "Expected full length 20000 for long balanced parentheses");
54  
55          // Stress test: no valid pairs
56          String onlyLeft = "(".repeat(30000);
57          assertEquals(0, app.longestValidParentheses(onlyLeft), "All '(' should return 0");
58  
59          String onlyRight = ")".repeat(30000);
60          assertEquals(0, app.longestValidParentheses(onlyRight), "All ')' should return 0");
61      }
62  
63      @Test
64      void generateParenthesis() {
65          // Example cases
66          assertEquals(List.of("()"), app.generateParenthesis(1), "n=1 should return ['()']");
67  
68          List<String> expectedN3 = List.of("((()))","(()())","(())()","()(())","()()()");
69          assertTrue(app.generateParenthesis(3).containsAll(expectedN3)
70                          && expectedN3.containsAll(app.generateParenthesis(3)),
71                  "n=3 should return all 5 well-formed combinations");
72  
73          // Edge cases
74          assertEquals(2, app.generateParenthesis(2).size(),
75                  "n=2 should generate exactly 2 combinations: (()) and ()()");
76          assertThrows(IllegalArgumentException.class, () -> app.generateParenthesis(0),
77                  "n=0 should throw exception");
78          assertThrows(IllegalArgumentException.class, () -> app.generateParenthesis(9),
79                  "n=9 should throw exception (exceeds constraint)");
80  
81          // Upper bound stress case (n=8) - just check size, not content
82          // Catalan number C8 = 1430
83          assertEquals(1430, app.generateParenthesis(8).size(),
84                  "n=8 should generate 1430 valid combinations");
85      }
