29      @Test
30      @DisplayName("Should calculate total portfolio value correctly with initial holdings")
31      void testGetTotalPortfolioValueInitial() {
32          // AAPL: 100 * 150 = 15000
33          // GOOGL: 50 * 2800 = 140000
34          // MSFT: 75 * 300 = 22500
35          double expected = 15000 + 140000 + 22500;
36          assertEquals(expected, portfolio.getTotalPortfolioValue(), DEFAULT_DELTA);
37      }

39      @Test
40      @DisplayName("Should update total portfolio value after buying new stock")
41      void testGetTotalPortfolioValueAfterBuy() {
42          portfolio.buyStock("AAPL", 10.0); // +10 * 150 = +1500
43          double expected = 15000 + 140000 + 22500 + 1500;
44          assertEquals(expected, portfolio.getTotalPortfolioValue(), DEFAULT_DELTA);
45      }

47      @Test
48      @DisplayName("Should add stock quantity correctly when buying existing stock")

52          assertEquals(expected, portfolio.getTotalPortfolioValue(), DEFAULT_DELTA);

55      @Test
56      @DisplayName("Should add new stock when buying non-existing stock")
57      void testBuyNewStock() {
58          portfolio.buyStock("TSLA", 5.0);
59          // TSLA has no defined price in calculateTotalValue, so it won't add to the value,
60          // but we can still verify holdings got updated (indirectly by selling it back to 0)
61          portfolio.sellStock("TSLA", 5.0);
62          // Should succeed because 5 was added
63          // Portfolio value should remain unchanged
64          double expected = 15000 + 140000 + 22500;
65          assertEquals(expected, portfolio.getTotalPortfolioValue(), DEFAULT_DELTA);

160     @Test
161     void runPortfolio() {
162         portfolio = new FinancialPortfolio();
163 
164         // Multiple readers
165         Runnable reader = () -> {
166             for (int i = 0; i < 5; i++) {
167                 double value = portfolio.getTotalPortfolioValue();
168                 if (DEBUG) {
169                     System.out.println(Thread.currentThread().getName() +
170                             " read value: $" + value);
171                 }
172                 try { Thread.sleep(100); } catch (InterruptedException e) {}
173             }
174         };
175 
176         // Writer
177         Runnable writer = () -> {
178             for (int i = 0; i < 3; i++) {
179                 portfolio.buyStock("AAPL", 10);
180                 try {
181                     Thread.sleep(200);
182                 } catch (InterruptedException e) {
183                     if (DEBUG) {
184                         System.out.println(e.getMessage());
185                     }
186                 }
187             }
188         };
189 
190         // Start threads
191         new Thread(reader, "Reader-1").start();
192         new Thread(reader, "Reader-2").start();
193         new Thread(writer, "Writer-1").start();
194         new Thread(reader, "Reader-3").start();
195 
196         assertTrue(portfolio.getTotalPortfolioValue() > 0);
197     }
