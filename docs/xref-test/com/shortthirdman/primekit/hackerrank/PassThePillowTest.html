3   import org.junit.jupiter.api.BeforeEach;
4   import org.junit.jupiter.api.Test;
5   
6   import static org.junit.jupiter.api.Assertions.*;
7   
8   class PassThePillowTest {
9   
10      PassThePillow app;
11  
12      @BeforeEach
13      void setUp() {
14          app = new PassThePillow();
15      }
16  
17      @Test
18      public void testNormalCases() {
19          assertEquals(2, app.pillowPass(4, 5));  // Example 1
20          assertEquals(3, app.pillowPass(3, 2));  // Example 2
21          assertEquals(2, app.pillowPass(2, 1));  // Simple Case
22          assertEquals(1, app.pillowPass(2, 2));  // Back to start
23          assertEquals(5, app.pillowPass(5, 4));  // Mid Line
24      }
25  
26      @Test
27      public void testEdgeCases() {
28          assertNotEquals(2, app.pillowPass(1000, 999));  // Large input edge case
29          assertEquals(1, app.pillowPass(2, 1000));    // Min `n`, Max `time`
30          assertEquals(501, app.pillowPass(1000, 500)); // Halfway through forward
31          assertThrows(IllegalArgumentException.class, () -> app.pillowPass(1000, 1998)); // Just before the end of cycle
32          assertThrows(IllegalArgumentException.class, () -> app.pillowPass(1000, 2000));  // Start of new cycle
33      }
34  
35      @Test
36      public void testInvalidInputs() {
37          assertThrows(IllegalArgumentException.class, () -> app.pillowPass(1, 100));
38          assertThrows(IllegalArgumentException.class, () -> app.pillowPass(1001, 100));
39          assertThrows(IllegalArgumentException.class, () -> app.pillowPass(100, 0));
40          assertThrows(IllegalArgumentException.class, () -> app.pillowPass(100, 1001));
41      }
42  }
