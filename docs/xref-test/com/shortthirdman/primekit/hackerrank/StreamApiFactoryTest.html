31      void concatenateEvenLengthWords() {
32          List<String> words = Arrays.asList("apple", "banana", "cherry", "date", "elderberry");
33          String actualWords = app.concatenateEvenLengthWords(words);
34          assertNotEquals("bananacherrydateelderberry", actualWords);
35          assertEquals("bananacherry", actualWords);
36          assertEquals("bananacherry".length(), actualWords.length());
37  
38          assertThrows(IllegalArgumentException.class, () -> app.concatenateEvenLengthWords(List.of()));
39          assertThrows(IllegalArgumentException.class, () -> app.concatenateEvenLengthWords(null));
40  
41          words = Arrays.asList("a", "abc", "def");
42          assertEquals("", app.concatenateEvenLengthWords(words));
43  
44          words = List.of("test");
45          assertEquals("test", app.concatenateEvenLengthWords(words));
46      }
