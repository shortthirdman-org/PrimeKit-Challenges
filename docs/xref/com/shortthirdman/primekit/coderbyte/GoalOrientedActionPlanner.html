16  /**
17   * A planner that uses the A* search algorithm to find an optimal sequence of actions
18   * (a Plan) to transition from an initial state to a desired goal state.
19   * @author shortthirdman
20   * @since 1.2.0
21   */
22  public class GoalOrientedActionPlanner {
23  
24      /**
25       * Calculates the optimal plan to reach the goal state from the initial state
26       * using the provided set of available actions.
27       * <p>
28       * This method implements the A* search algorithm, exploring states based on
29       * accumulated cost (g-score) and a heuristic estimating the remaining distance
30       * to the goal (f-score).
31       *
32       * @param initialState     the starting state map containing key-value condition pairs
33       * @param goal             the target state map containing required key-value condition pairs
34       * @param availableActions the list of actions the planner can utilize
35       * @return an {@link Optional} containing the optimal {@link Plan} if a valid sequence
36       * is found, or an empty {@link Optional} if the goal is unreachable
37       */
38      public Optional<Plan> plan(
39              Map<String, Object> initialState,
40              Map<String, Object> goal,
41              List<Action> availableActions) {
42          var open = new PriorityQueue<>(Comparator.comparingDouble(Node::f));
43          var closed = new HashSet<Map<String, Object>>();
44  
45          open.add(new Node(initialState, List.of(), 0, heuristic(initialState, goal)));
46  
47          while (!open.isEmpty()) {
48              Node current = open.poll();
49  
50              if (satisfies(current.state(), goal)) {
51                  return Optional.of(new Plan(current.actions(), current.g()));
52              }
53  
54              if (!closed.add(current.state())) continue;
55  
56              for (Action action : availableActions) {
57                  if (!canApply(action, current.state())) continue;
58  
59                  Map<String, Object> next = apply(action, current.state());
60                  List<Action> chain = new ArrayList<>(current.actions());
61                  chain.add(action);
62  
63                  double g = current.g() + action.cost();
64                  double f = g + heuristic(next, goal);
65  
66                  open.add(new Node(next, chain, g, f));
67              }
68          }
69  
70          return Optional.empty();
71      }
72  
73      /**
74       * Evaluates whether an action can be performed in the given state by verifying
75       * that all preconditions of the action are met.
76       *
77       * @param action the action to evaluate
78       * @param state  the current state to check against
79       * @return {@code true} if all preconditions are satisfied, {@code false} otherwise
80       */
81      private boolean canApply(Action action, Map<String, Object> state) {
82          return action.preconditions().entrySet().stream()
83                  .allMatch(e -> Objects.equals(state.get(e.getKey()), e.getValue()));
84      }
85  
86      /**
87       * Applies the effects of an action to a given state, generating a new resulting state.
88       * <p>
89       * This method does not mutate the original state; instead, it returns a newly allocated
90       * map containing the merged state and effects.
91       *
92       * @param action the action whose effects are to be applied
93       * @param state  the initial state before the action is executed
94       * @return a new state map reflecting the applied effects
95       */
96      private Map<String, Object> apply(Action action, Map<String, Object> state) {
97          Map<String, Object> result = new HashMap<>(state);
98          result.putAll(action.effects());
99          return result;
100     }
101 
102     /**
103      * Checks if the given state satisfies all conditions defined in the goal.
104      *
105      * @param state the current state to evaluate
106      * @param goal  the target conditions that must be met
107      * @return {@code true} if the state contains all goal conditions with matching values,
108      * {@code false} otherwise
109      */
110     private boolean satisfies(Map<String, Object> state, Map<String, Object> goal) {
111         return goal.entrySet().stream()
112                 .allMatch(e -> Objects.equals(state.get(e.getKey()), e.getValue()));
113     }
114 
115     /**
116      * Calculates a heuristic value estimating the cost to reach the goal from the current state.
117      * <p>
118      * The current implementation uses the number of unsatisfied goal conditions as the heuristic.
119      * This acts as an admissible heuristic assuming each action costs at least 1.0 and satisfies
120      * at most one goal condition.
121      *
122      * @param state the current state
123      * @param goal  the target state
124      * @return the count of goal conditions that differ from the current state
125      */
126     private double heuristic(Map<String, Object> state, Map<String, Object> goal) {
127         return goal.entrySet().stream()
128                 .filter(e -> !Objects.equals(state.get(e.getKey()), e.getValue()))
129                 .count();
130     }
131 
132     /**
133      * Represents a discrete node in the A* search tree.
134      *
135      * @param state   the current world state at this node
136      * @param actions the sequence of actions taken to reach this state
137      * @param g       the exact cumulative cost from the initial state to this node (g-score)
138      * @param f       the estimated total cost from the initial state to the goal through this node (f-score = g + heuristic)
139      */
140     private record Node(
141             Map<String, Object> state,
142             List<Action> actions,
143             double g,
144             double f
145     ) {}
146 }
