10  import java.util.Objects;
11  
12  /**
13   * Calculates tax by category and strategy
14   * @since 1.1.1
15   * @author ShortThirdMan
16   */
17  public class PatternAnalyzer {
18  
19      private static final Boolean DEBUG = System.getenv("DEBUG") != null || System.getProperty("DEBUG") != null;
20  
21      public double calculateTaxByStrategy(double baseAmount, ProductCategory category) {
22          if (Objects.isNull(category)) {
23              throw new IllegalArgumentException("Category is null");
24          }
25  
26          TaxCalculationStrategy strategy = switch (category) {
27              case STANDARD -> new StandardTaxCalculation();
28              case PREMIUM -> new PremiumTaxCalculation();
29              case LUXURY -> new LuxuryTaxCalculation();
30              default -> throw new IllegalArgumentException("Unsupported product category");
31          };
32  
33          double taxAmount = strategy.calculateTax(baseAmount);
34          if (DEBUG) {
35              System.out.println("[CalculateTaxByStrategy] Tax for $" + baseAmount + " (" + category + "): $" + taxAmount);
36          }
37  
38          return taxAmount;
39      }
40  
41      public double calculateTaxByCategory(double baseAmount, ProductCategory category) {
42          if (Objects.isNull(category)) {
43              throw new IllegalArgumentException("Category is null");
44          }
45  
46          TaxCalculator calculator = switch (category) {
47              case STANDARD -> amount1 -> amount1 * 0.10; // 10% tax
48              case PREMIUM -> amount1 -> amount1 * 0.20; // 20% tax
49              case LUXURY -> amount1 -> amount1 * 0.30; // 30% tax
50              default -> throw new IllegalArgumentException("Unsupported product category");
51          };
52  
53          double taxAmount = calculator.calculate(baseAmount);
54          if (DEBUG) {
55              System.out.println("[CalculateTaxByCategory] Tax for $" + baseAmount + " (" + category + "): $" + taxAmount);
56          }
57  
58          return taxAmount;
59      }
60  }
