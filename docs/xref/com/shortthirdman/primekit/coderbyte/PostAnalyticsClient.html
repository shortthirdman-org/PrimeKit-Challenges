3   import com.google.gson.Gson;
4   import com.google.gson.JsonArray;
5   import com.google.gson.reflect.TypeToken;

8   import java.net.URI;
9   import java.net.http.HttpClient;
10  import java.net.http.HttpRequest;
11  import java.net.http.HttpResponse;
12  import java.util.LinkedHashMap;
13  import java.util.List;
14  import java.util.Map;
15  import java.util.stream.Collectors;
16  
17  public class PostAnalyticsClient {
18  
19      private static final String ALL_POSTS = "https://coderbyte.com/api/challenges/json/all-posts";
20  
21      /**
22       * Retrieve a list of posts.
23       * @return list of {@link Post}
24       */
25      public static List<Post> fetchPosts() {
26          try (HttpClient client = HttpClient.newHttpClient()) {
27              HttpRequest request = HttpRequest.newBuilder()
28                      .uri(URI.create(ALL_POSTS))
29                      .GET()
30                      .build();
31  
32              HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
33  
34              Gson gson = new Gson();
35  
36              // Response is a JSON ARRAY
37              JsonArray jsonArray = gson.fromJson(response.body(), JsonArray.class);
38  
39              return gson.fromJson(
40                      jsonArray,
41                      new TypeToken<List<Post>>() {}.getType()
42              );
43  
44          } catch (Exception e) {
45              throw new RuntimeException("Failed to fetch posts", e);
46          }
47      }
48  
49      /**
50       * Summary of how many posts each user has made
51       * @param posts the input list of posts
52       * @return a list of objects each containing only the keys userId and number of {@link Post}s with userId appearing first.
53       */
54      public static List<Map<String, Integer>> formatPosts(List<Post> posts) {
55          if (posts == null || posts.isEmpty()) {
56              throw new IllegalArgumentException("No posts provided to format");
57          }
58  
59          return posts.stream()
60                  .collect(Collectors.groupingBy(
61                          Post::userId,
62                          Collectors.collectingAndThen(Collectors.counting(), Long::intValue)
63                  ))
64                  .entrySet()
65                  .stream()
66                  .sorted((a, b) -> {
67                      int cmp = Integer.compare(b.getValue(), a.getValue());
68                      return (cmp != 0) ? cmp : Integer.compare(b.getKey(), a.getKey());
69                  })
70                  .map(entry -> {
71                      Map<String, Integer> map = new LinkedHashMap<>();
72                      map.put("userId", entry.getKey());
73                      map.put("numberOfPosts", entry.getValue());
74                      return map;
75                  })
76                  .toList();
77      }
78  
79      public static List<Map<String, Integer>> fetchAndFormat() {
80          List<Post> posts = fetchPosts();
81          return formatPosts(posts);
82      }
83  }
