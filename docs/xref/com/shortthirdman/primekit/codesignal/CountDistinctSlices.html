1   package com.shortthirdman.primekit.codesignal;

3   import java.util.Arrays;
4   
5   /**
6    * An integer M and a non-empty array A consisting of N non-negative integers are given.
7    * All integers in array A are less than or equal to <code>M</code>
8    * A pair of integers <code>(P, Q)</code>, such that <code>0 &le; P &le; Q &lt; N</code>, is called a slice of array <code>A</code>.
9    * <br></br>
10   * The slice consists of the elements A[P], A[P + 1], ..., A[Q]. A distinct slice is a slice consisting of only unique numbers.
11   * That is, no individual number occurs more than once in the slice.
12   * <br></br>
13   * For example, consider integer M = 6 and array A such that:
14   * <pre>
15   *     A[0] = 3
16   *     A[1] = 4
17   *     A[2] = 5
18   *     A[3] = 5
19   *     A[4] = 2
20   * </pre>
21   * There are exactly nine distinct slices:
22   * <pre>(0, 0), (0, 1), (0, 2), (1, 1), (1, 2), (2, 2), (3, 3), (3, 4) and (4, 4).</pre>
23   *
24   * The goal is to calculate the number of distinct slices.
25   *
26   * @author ShortThirdMan
27   */
28  @SuppressWarnings("PMD.UselessParentheses")
29  public class CountDistinctSlices {
30  
31      /**
32       * Given an integer M and a non-empty array A consisting of N integers, returns the number of distinct slices.
33       * <br>
34       * If the number of distinct slices is greater than 1,000,000,000, the function should return 1,000,000,000.
35       * @param M integer
36       * @param A non-empty array
37       * @return number of distinct slices
38       */
39      public int calculateDistinctSlices(int M, int[] A) {
40          int N = A.length;
41          if (N < 1 || N > 100000) {
42              throw new IllegalArgumentException("Array length should be between 1 and 100000");
43          }
44  
45          if (M < 0 || M > 100000) {
46              throw new IllegalArgumentException("Integer M should be between 0 and 100000");
47          }
48  
49          boolean isNotInRange = Arrays.stream(A).boxed().anyMatch(n -> n < 0 || n > M);
50          if (isNotInRange) {
51              throw new IllegalArgumentException("Each element in array should be between 0 and " + M);
52          }
53  
54          int[] lastSeen = new int[M + 1]; // To track the last occurrence of each element
55          int left = 0;
56          long distinctSlices = 0;
57          final int MOD = 1_000_000_007;
58  
59          // Traverse with the right pointer
60          for (int right = 0; right < N; right++) {
61              // If the current element has appeared before in the window, move 'left'
62              lastSeen[A[right]]++;
63  
64              // Move the 'left' pointer until the slice [left, right] is distinct
65              while (lastSeen[A[right]] > 1) {
66                  lastSeen[A[left]]--;
67                  left++;
68              }
69  
70              // Add the number of distinct slices ending at 'right'
71              var windowSize = right - left + 1;
72              distinctSlices = (distinctSlices + windowSize) % MOD;
73          }
74  
75          // Return the result (casting to int)
76          return (int) distinctSlices;
77      }
78  }
