6   public class FrameGenerator {
7   
8       /**
9        * Generates a square frame of size {@code n} as an array of strings.
10       * @param n the size of the square frame; must be at least 2
11       * @return an array of {@code n} strings, each of length {@code n}, representing the frame
12       * @throws IllegalArgumentException if {@code n < 2}
13       */
14      public String[] generate(int n) {
15          if (n < 2) {
16              throw new IllegalArgumentException("Frame size must be at least 2.");
17          }
18  
19          String[] frame = new String[n];
20  
21          // Top and bottom rows are completely filled with '*'
22          String borderRow = "*".repeat(n);
23  
24          // Middle rows have '*' at start and end, spaces in between
25          String middleRow = "*" + " ".repeat(n - 2) + "*";
26  
27          for (int i = 0; i < n; i++) {
28              if (i == 0 || i == n - 1) {
29                  frame[i] = borderRow;
30              } else {
31                  frame[i] = middleRow;
32              }
33          }
34  
35          return frame;
36      }
37  }
