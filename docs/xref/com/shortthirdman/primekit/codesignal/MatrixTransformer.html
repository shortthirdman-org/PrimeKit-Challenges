3   /**
4    * In a highly automated warehouse, a robot is tasked with organizing packages stored in a rectangular grid.
5    * The grid is represented as a 2D list of integers matrix, where each integer represents a package type.
6    * To manage the sorting and rearrangement, the robot follows a series of commands given as an array of strings commands.<br/>
7    * Each command instructs the robot to perform a specific operation on the matrix:
8    * <br/>
9    *     "swapRows r1 r2": swaps the elements of row r1 with the elements of row r2.<br/>
10   *     "swapColumns c1 c2": swaps the elements of column c1 with the elements of column c2.<br/>
11   *     "reverseRow r": reverses the elements of row r.<br/>
12   *     "reverseColumn c": reverses the elements of column c.<br/>
13   *     "rotate90Clockwise": rotates the entire matrix 90 degrees clockwise.<br/>
14   * <br/>
15   * Here, r, r1, and r2 are integer row indices, and c, c1, and c2 are integer column indices.
16   * @author ShortThirdMan
17   */
18  public class MatrixTransformer {
19  
20      /**
21       * Transform matrics with given commands
22       * @param matrix the input matrix
23       * @param commands the commands given to robot
24       * @return
25       */
26      public int[][] transformMatricesWithCommands(int[][] matrix, String[] commands) {
27          for (String command : commands) {
28              String[] parts = command.split(" ");
29              switch (parts[0]) {
30                  case "swapRows":
31                      int r1 = Integer.parseInt(parts[1]);
32                      int r2 = Integer.parseInt(parts[2]);
33                      int[] tempRow = matrix[r1];
34                      matrix[r1] = matrix[r2];
35                      matrix[r2] = tempRow;
36                      break;
37  
38                  case "swapColumns":
39                      int c1 = Integer.parseInt(parts[1]);
40                      int c2 = Integer.parseInt(parts[2]);
41                      for (int i = 0; i < matrix.length; i++) {
42                          int temp = matrix[i][c1];
43                          matrix[i][c1] = matrix[i][c2];
44                          matrix[i][c2] = temp;
45                      }
46                      break;
47  
48                  case "reverseRow":
49                      int r = Integer.parseInt(parts[1]);
50                      int cols = matrix[r].length;
51                      for (int i = 0; i < cols / 2; i++) {
52                          int temp = matrix[r][i];
53                          matrix[r][i] = matrix[r][cols - 1 - i];
54                          matrix[r][cols - 1 - i] = temp;
55                      }
56                      break;
57  
58                  case "reverseColumn":
59                      int c = Integer.parseInt(parts[1]);
60                      int rows = matrix.length;
61                      for (int i = 0; i < rows / 2; i++) {
62                          int temp = matrix[i][c];
63                          matrix[i][c] = matrix[rows - 1 - i][c];
64                          matrix[rows - 1 - i][c] = temp;
65                      }
66                      break;
67  
68                  case "rotate90Clockwise":
69                      matrix = rotate90Clockwise(matrix);
70                      break;
71              }
72          }
73          return matrix;
74      }
75  
76      private  int[][] rotate90Clockwise(int[][] matrix) {
77          int m = matrix.length;
78          int n = matrix[0].length;
79          int[][] rotated = new int[n][m];
80          for (int i = 0; i < m; i++) {
81              for (int j = 0; j < n; j++) {
82                  rotated[j][m - 1 - i] = matrix[i][j];
83              }
84          }
85          return rotated;
86      }
