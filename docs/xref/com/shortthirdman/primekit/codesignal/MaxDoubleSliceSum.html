1   package com.shortthirdman.primekit.codesignal;
2   
3   import java.util.Arrays;
4   
5   /**
6    *
7    * Find the maximal sum of any double slice.
8    * <br><br/>
9    * A non-empty array A consisting of N integers is given.
10   * <br><br/>
11   * <pre>
12   * A triplet (X, Y, Z), such that 0 &le; X &lt; Y &lt; Z &lt; N, is called a double slice.
13   * </pre>
14   * A double slice sum is the sum of the elements of the array A between indices
15   * X and Y (exclusive) and between indices Y and Z (exclusive):
16   * <br><br/>
17   * The sum of double slice (X, Y, Z) is the total of A[X + 1] + A[X + 2] + ... + A[Y − 1] + A[Y + 1] + A[Y + 2] + ... + A[Z − 1].
18   *
19   * @author ShortThirdMan
20   */
21  public class MaxDoubleSliceSum {
22  
23      public int calculateMaxDoubleSum(int[] A) {
24          int N = A.length;
25          if (N < 3 || N > 100000) {
26              throw new IllegalArgumentException("Array length must be between 3 and 100,000.");
27          }
28  
29          boolean flag = Arrays.stream(A).boxed().allMatch(num -> num >= -10000 && num <= 10000);
30          if (!flag) {
31              throw new IllegalArgumentException("Array elements must be between -10,000 and 10,000.");
32          }
33  
34          // Step 1: Calculate left max slice sums
35          int[] leftMax = new int[N];
36          int maxEndingHere = 0;
37          for (int i = 1; i < N; i++) {
38              maxEndingHere = Math.max(0, maxEndingHere + A[i]);
39              leftMax[i] = maxEndingHere;
40          }
41  
42          // Step 2: Calculate right max slice sums
43          int[] rightMax = new int[N];
44          maxEndingHere = 0;
45          for (int i = N - 2; i >= 0; i--) {
46              maxEndingHere = Math.max(0, maxEndingHere + A[i]);
47              rightMax[i] = maxEndingHere;
48          }
49  
50          // Step 3: Find the maximum sum of any double slice
51          int maxDoubleSliceSum = 0;
52          for (int i = 1; i < N - 1; i++) {
53              maxDoubleSliceSum = Math.max(maxDoubleSliceSum, leftMax[i - 1] + rightMax[i + 1]);
54          }
55  
56          return maxDoubleSliceSum;
57      }
58  }
