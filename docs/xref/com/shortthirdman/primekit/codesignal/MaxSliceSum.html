3   /**
4    * A non-empty array A consisting of N integers is given. A pair of integers (P, Q), such that <code>0 &le; P &le; Q &lt; N</code>,
5    * is called a slice of array <code>A</code>.<br/>
6    * The sum of a slice <code>(P, Q)</code> is the total of <code>A[P] + A[P+1] + ... + A[Q]</code>.
7    * Write a function:
8    * that, given an array A consisting of N integers, returns the maximum sum of any slice of A.
9    * Find a maximum sum of a compact subsequence of array elements.
10   *
11   * @author ShortThirdMan
12   */
13  public class MaxSliceSum {
14  
15      public int maxSumForArray(int[] A) {
16          if (A == null || A.length == 0) {
17              throw new IllegalArgumentException("Array cannot be null or empty");
18          }
19  
20          int N = A.length;
21          if (N == 1) {
22              return A[0];
23          }
24  
25          int maxCurrent = A[0];
26          int maxGlobal = A[0];
27  
28          for (int i = 1; i < N; i++) {
29              maxCurrent = Math.max(A[i], maxCurrent + A[i]);
30              maxGlobal = Math.max(maxGlobal, maxCurrent);
31          }
32  
33          return maxGlobal;
34      }
35  }
