1   package com.shortthirdman.primekit.codesignal;
2   
3   import java.util.ArrayList;
4   import java.util.List;
5   
6   /**
7    * You are moderating a newspaper page, and you have to align the text on the page properly.
8    * The text is provided to you in the following format:
9    * <br/>
10   * <p><b>paragraphs</b> is an array of paragraphs, where each paragraph is represented as an array of words;</p>
11   * <p><b>aligns</b> is an array representing the alignment of each paragraph from paragraphs - each element is either "LEFT" or "RIGHT";</p>
12   * <p><b>width</b> represents the maximum number of characters each line of the output can include.</p>
13   * <br/>
14   * Your task is to produce a newspaper page according to the following specifications:
15   * <br/>
16   * <p>For each paragraph paragraphs[i], include all the words paragraphs[i][j] in order, separated by spaces;</p>
17   *
18   * <p>Include as many words as possible per each page line (the length of the line must be less than or equal to width),
19   * and put the next word on a new line if it would exceed the limit;</p>
20   *
21   * <p>In the case of excess whitespace, words from paragraphs[i] should be aligned according to aligns[i] - if aligns[i] = "LEFT",
22   * the line should have trailing spaces, if aligns[i] = RIGHT, it should have leading spaces;</p>
23   *
24   * <p>Include a border of * characters around all the edges of the result - these characters don't count toward the width,
25   * they are just added to make output more pretty.</p>
26   *
27   * <p>It is guaranteed that it is possible to justify the given paragraphs to the newspaper.</p>
28   * <br/>
29   * Return the resulting newspaper page as an array of strings.
30   *
31   * @author ShortThirdMan
32   */
33  public class NewspaperFormatter {
34  
35      /**
36       * @param paragraphs the input array of paragraphs
37       * @param aligns the alignment array
38       * @param width the width
39       * @return the array of formatted strings
40       */
41      public String[] formatPage(String[][] paragraphs, String[] aligns, int width) {
42          if (width < 5 || width > 100) {
43              throw new StringIndexOutOfBoundsException("width should be between 5 and 100");
44          }
45  
46          List<String> result = new ArrayList<>();
47          String border = "*".repeat(width + 2);
48          result.add(border);
49  
50          for (int i = 0; i < paragraphs.length; i++) {
51              String[] words = paragraphs[i];
52              String align = aligns[i];
53              List<String> lines = new ArrayList<>();
54              StringBuilder lineBuilder = new StringBuilder();
55  
56              for (String word : words) {
57                  if (lineBuilder.isEmpty()) {
58                      lineBuilder.append(word);
59                  } else if (lineBuilder.length() + 1 + word.length() <= width) {
60                      lineBuilder.append(" ").append(word);
61                  } else {
62                      lines.add(lineBuilder.toString());
63                      lineBuilder = new StringBuilder(word);
64                  }
65              }
66  
67              if (!lineBuilder.isEmpty()) {
68                  lines.add(lineBuilder.toString());
69              }
70  
71              for (String line : lines) {
72                  int spaces = width - line.length();
73                  String formatted;
74                  if ("LEFT".equals(align)) {
75                      formatted = line + " ".repeat(spaces);
76                  } else { // RIGHT
77                      formatted = " ".repeat(spaces) + line;
78                  }
79                  result.add("*" + formatted + "*");
80              }
81          }
82  
83          result.add(border);
84          return result.toArray(new String[0]);
85      }
86  }
