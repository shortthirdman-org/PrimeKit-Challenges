3   import java.util.HashMap;
4   import java.util.List;
5   import java.util.Map;
6   import java.util.function.Function;
7   import java.util.stream.Collectors;

9   /**
10   * You are working with data collected from various sensors. Given an array of non-negative integers readings representing the sensor readings,
11   * transform the array by repeatedly replacing each element with the sum of its digits. Continue this transformation until
12   * every element is a single digit. Return the most occurring digit in the final array. In case of a tie, return the highest digit.
13   *
14   * @author ShortThirdMan
15   */
16  public class SensorDigitAnalyzer {
17  
18      /**
19       * @param readings the energy power readings
20       * @return the most frequent digit
21       */
22      public int findMostFrequentSingleDigit(int[] readings) {
23          if (readings == null || readings.length == 0) {
24              return -1;
25          }
26  
27          Map<Integer, Integer> freqMap = new HashMap<>();
28  
29          for (int reading : readings) {
30              int singleDigit = reduceToSingleDigit(reading);
31              freqMap.put(singleDigit, freqMap.getOrDefault(singleDigit, 0) + 1);
32          }
33  
34          int maxFrequency = 0;
35          int mostFrequentDigit = -1;
36  
37          for (int digit = 0; digit <= 9; digit++) {
38              int frequency = freqMap.getOrDefault(digit, 0);
39              if (frequency > maxFrequency || (frequency == maxFrequency && digit > mostFrequentDigit)) {
40                  maxFrequency = frequency;
41                  mostFrequentDigit = digit;
42              }
43          }
44  
45          return mostFrequentDigit;
46      }
47  
48      private int reduceToSingleDigit(int num) {
49          while (num >= 10) {
50              num = String.valueOf(num)
51                      .chars()
52                      .map(Character::getNumericValue)
53                      .sum();
54          }
55          return num;
56      }
57  
58      public int findMostFrequentSingleDigit(List<Integer> readings) {
59          Map<Integer, Long> freqMap = readings.stream()
60                  .map(this::reduceToSingleDigit)
61                  .collect(Collectors.groupingBy(Function.identity(), Collectors.counting()));
62  
63          return freqMap.entrySet()
64                  .stream()
65                  .sorted((a, b) -> {
66                      int freqCompare = Long.compare(b.getValue(), a.getValue());
67                      if (freqCompare != 0) {
68                          return freqCompare;
69                      } else {
70                          return Integer.compare(b.getKey(), a.getKey());
71                      }
72                  })
73                  .findFirst()
74                  .map(Map.Entry::getKey)
75                  .orElse(-1);
76      }
77  }
