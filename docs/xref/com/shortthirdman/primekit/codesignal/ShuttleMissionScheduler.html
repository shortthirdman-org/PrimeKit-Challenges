1   package com.shortthirdman.primekit.codesignal;
2   
3   import java.util.Arrays;
4   import java.util.function.Predicate;
5   
6   /**
7    * A space station network has two major hubs, Alpha and Beta, connected by shuttle services.
8    * Shuttles from Alpha to Beta take 100 time units and depart according to a sorted integer array alpha2beta.
9    * Similarly, shuttles from Beta to Alpha also take 100 time units and depart at times specified in a sorted integer array beta2alpha.
10   * <br/>
11   * You need to complete missions where each mission requires traveling from Alpha to Beta and back to Alpha.
12   * Calculate the time unit when you will complete all missions, assuming you always take the earliest available shuttle.
13   * <br/>
14   * It is guaranteed that it is possible to complete all missions using the provided shuttle schedules.
15   * <br/>
16   * All times are measured in time units from the start of your journey.
17   *
18   * @author ShortThirdMan
19   */
20  public class ShuttleMissionScheduler {
21  
22      /**
23       * @param alpha2beta the time taken from Alpha to Beta shuttle
24       * @param beta2alpha the time taken from Beta to Alpha shuttle
25       * @param missions the number of mission
26       * @return
27       */
28      public int scheduleShuttle(int[] alpha2beta, int[] beta2alpha, int missions) {
29          if (alpha2beta.length == 0 || beta2alpha.length == 0) {
30              return -1;
31          }
32  
33          if (missions < 1 || missions > 10) {
34              throw new IllegalArgumentException("missions must be between 1 and 10");
35          }
36  
37          if (alpha2beta.length > 100 ||  beta2alpha.length > 100) {
38              throw new IllegalArgumentException("alpha2beta and beta2alpha length should be between 1 and 100");
39          }
40  
41          Predicate<int[]> isValidArray = arr -> Arrays.stream(arr).allMatch(x -> x >= 0 && x <= 2000);
42  
43          if (!isValidArray.test(alpha2beta) || !isValidArray.test(beta2alpha)) {
44              throw new IllegalArgumentException("alpha2beta and beta2alpha elements should be between 0 and 2000");
45          }
46  
47          Arrays.sort(alpha2beta);
48          Arrays.sort(beta2alpha);
49  
50          int currentTime = 0;
51  
52          for (int i = 0; i < missions; i++) {
53              // Step 1: Take the earliest shuttle from Alpha to Beta
54              int depAlphaToBeta = findNextShuttle(alpha2beta, currentTime);
55              int arriveAtBeta = depAlphaToBeta + 100;
56  
57              // Step 2: Take the earliest shuttle from Beta to Alpha
58              int depBetaToAlpha = findNextShuttle(beta2alpha, arriveAtBeta);
59              currentTime = depBetaToAlpha + 100; // back at Alpha
60          }
61  
62          return currentTime;
63      }
64  
65      private int findNextShuttle(int[] schedule, int currentTime) {
66          int left = 0, right = schedule.length - 1;
67          int result = -1;
68  
69          while (left <= right) {
70              int mid = left + (right - left) / 2;
71              if (schedule[mid] >= currentTime) {
72                  result = schedule[mid];
73                  right = mid - 1;
74              } else {
75                  left = mid + 1;
76              }
77          }
78  
79          return result;
80      }
81  }
