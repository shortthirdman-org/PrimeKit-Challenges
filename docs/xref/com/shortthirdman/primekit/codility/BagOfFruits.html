3   import java.util.ArrayList;
4   import java.util.HashMap;
5   import java.util.List;
6   import java.util.Map;
7   import java.util.function.Function;
8   import java.util.stream.Collectors;

10  public class BagOfFruits {
11  
12      public List<String> countFruits(List<String> fruits, long threshold) {
13          Map<String, Long> countingMap = fruits.stream()
14                  .collect(Collectors.groupingBy(Function.identity(),
15                          Collectors.counting()));
16  
17          return countingMap.entrySet()
18                  .stream()
19                  .filter(a -> a.getValue() > threshold)
20                  .map(Map.Entry::getKey)
21                  .collect(Collectors.toList());
22      }
23  
24      public List<String> countFruits(List<String> fruits, int threshold) {
25          Map<String, Integer> countingMap = new HashMap<>();
26          for (String fruit : fruits) {
27              countingMap.put(fruit, countingMap.getOrDefault(fruit, 0) + 1);
28          }
29  
30          List<String> result = new ArrayList<>();
31          for (Map.Entry<String, Integer> fruitCountMap : countingMap.entrySet()) {
32              if (fruitCountMap.getValue() > threshold) {
33                  result.add(fruitCountMap.getKey());
34              }
35          }
36  
37          return result;
38      }
39  }
