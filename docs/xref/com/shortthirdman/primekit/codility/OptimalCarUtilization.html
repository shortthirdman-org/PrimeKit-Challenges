1   package com.shortthirdman.primekit.codility;
2   
3   import java.util.Arrays;
4   import java.util.Collections;
5   
6   /**
7    * A group of friends is going on holiday together. They have come to a meeting point (the start of the journey)
8    * using N cars. There are <code>P[K]</code> people and <code>S[K]</code> seats in the K-th car for K in range <code>[0...N-1]</code>.
9    * Some of the seats in the cars may be free, so it is possible for some of the friends to change the car they are in.
10   * The friends have decided that, in order to be ecological, they will leave some cars parked at the meeting point
11   * and travel with as few cars as possible.
12   *
13   * @author ShortThirdMan
14   * @since 1.0.0
15   */
16  public class OptimalCarUtilization {
17  
18      public int optimizeCarsUsed(int[] P, int[] S) {
19          int N = P.length;
20          int totalPeople = 0;
21          Integer[] seats = new Integer[N];
22          int totalSeats = 0;
23  
24          // Calculate the total number of people and prepare the seats array
25          for (int i = 0; i < N; i++) {
26              totalPeople += P[i];
27              seats[i] = S[i];
28              totalSeats += S[i];
29          }
30  
31          if (totalPeople > totalSeats) {
32              throw new IllegalStateException("There are more people than available seats.");
33          }
34  
35          if (totalPeople == 0) {
36              return 0;
37          }
38  
39          // Sort seats in descending order
40          Arrays.sort(seats, Collections.reverseOrder());
41  
42          int carsUsed = 0;
43          int accumulatedSeats = 0;
44  
45          // Use the cars with the most seats first
46          for (int i = 0; i < N; i++) {
47              accumulatedSeats += seats[i];
48              carsUsed++;
49              if (accumulatedSeats >= totalPeople) {
50                  break;
51              }
52          }
53  
54          return carsUsed;
55      }
56  }
