3   import java.util.HashSet;
4   import java.util.Set;

6   /**
7    * You are given a string consisting of lowercase letters of the English alphabet.
8    * You must split this string into a minimal number of substrings in such a way that no letter
9    * occurs more than once in each substring.
10   * For example, here are some correct splits of the string <pre>"abacdec"</pre>
11   * <pre>
12   *     ('a', 'bac', 'dec'), ('a', bacd', 'ec') and ('ab', 'ac', 'dec')
13   * </pre>
14   *
15   * @author ShortThirdMan
16   * @since 1.0.0
17   */
18  public class SubstringSplitter {
19  
20      public int splitSubstrings(String S) {
21          if (S == null) {
22              throw new NullPointerException("String input can not be null or empty");
23          }
24  
25          if (S.isEmpty()) {
26              return 0;
27          }
28  
29          Set<Character> seenCharacters = new HashSet<>();
30          int substringsCount = 1;  // We start with one substring
31  
32          for (char c : S.toCharArray()) {
33              // If the character is already in the current substring, start a new substring
34              if (seenCharacters.contains(c)) {
35                  substringsCount++;  // Start a new substring
36                  seenCharacters.clear();  // Clear the set for the new substring
37              }
38              // Add the current character to the set
39              seenCharacters.add(c);
40          }
41  
42          return substringsCount;
43      }
44  
45      @Deprecated
46      public int splitString(String S) {
47          int n = S.length();
48          if (n == 0) return 0;
49  
50          // Step 1: Find the last occurrence of each character
51          int[] lastOccurrence = new int[26];
52          for (int i = 0; i < n; i++) {
53              lastOccurrence[S.charAt(i) - 'a'] = i;
54          }
55  
56          // Step 2: Split the string into minimal substrings
57  //        int start = 0;
58          int end = 0; // The end of the current substring
59          int substringsCount = 0;
60  
61          for (int i = 0; i < n; i++) {
62              // Update the end of the current substring to be the farthest
63              // we need to go to include all characters seen so far
64              end = Math.max(end, lastOccurrence[S.charAt(i) - 'a']);
65  
66              // When we reach the end of the current substring
67              if (i == end) {
68                  substringsCount++;
69  //                start = i + 1;
70                  end = i + 1; // Update end to start a new substring
71              }
72          }
73  
74          return substringsCount;
75      }
76  }
