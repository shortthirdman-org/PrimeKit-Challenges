12  /**
13   * Given a string <code>s</code> and a dictionary of strings <code>wordDict</code>, add spaces in s to construct a sentence
14   * where each word is a valid dictionary word. Return all such possible sentences in any order.
15   * Note that the same word in the dictionary may be reused multiple times in the segmentation.
16   * <br/>
17   * <b>Example 1</b>
18   *
19   * Input: s = "catsanddog", wordDict = ["cat","cats","and","sand","dog"]
20   * Output: ["cats and dog","cat sand dog"]
21   * <br/>
22  
23   * <b>Example 2</b>
24   * Input: s = "pineapplepenapple", wordDict = ["apple","pen","applepen","pine","pineapple"]
25   * Output: ["pine apple pen apple","pineapple pen apple","pine applepen apple"]
26   * Explanation: Note that you are allowed to reuse a dictionary word.
27   * <br/>
28  
29   * <b>Example 3</b>
30   * Input: s = "catsandog", wordDict = ["cats","dog","sand","and","cat"]
31   * Output: []
32  
33   * @apiNote <a href="https://leetcode.com/problems/word-break-ii/">Word Break II</a>
34  
35   * @author ShortThirdMan
36   */
37  public class WordBreak {
38  
39      public List<String> wordBreak(String s, List<String> wordDict) {
40          // Convert the word dictionary to a set for O(1) lookups
41          Set<String> wordSet = new HashSet<>(wordDict);
42          // Memoization map
43          Map<Integer, List<String>> memo = new HashMap<>();
44          return backtrack(s, 0, wordSet, memo);
45      }
46  
47      private List<String> backtrack(String s, int start, Set<String> wordSet, Map<Integer, List<String>> memo) {
48          // If we reach the end of the string
49          if (start == s.length()) {
50              return Collections.singletonList("");
51          }
52  
53          // Check if the result is already in the memo
54          if (memo.containsKey(start)) {
55              return memo.get(start);
56          }
57  
58          List<String> sentences = new ArrayList<>();
59  
60          // Try every possible end index
61          for (int end = start + 1; end <= s.length(); end++) {
62              String word = s.substring(start, end);
63              if (wordSet.contains(word)) {
64                  // Recursively break the rest of the string
65                  List<String> restSentences = backtrack(s, end, wordSet, memo);
66                  for (String sentence : restSentences) {
67                      if (sentence.isEmpty()) {
68                          sentences.add(word);
69                      } else {
70                          sentences.add(word + " " + sentence);
71                      }
72                  }
73              }
74          }
75  
76          // Memorize the result for this start index
77          memo.put(start, sentences);
78          return sentences;
79      }
80  }
