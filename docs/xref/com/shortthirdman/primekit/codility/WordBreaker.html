1   package com.shortthirdman.primekit.codility;

3   import java.util.HashSet;
4   import java.util.List;
5   import java.util.Set;
6   
7   /**
8    * Given a string s and a dictionary of strings wordDict,
9    * return true if s can be broken into one or more dictionary words.
10   * Note that the same word in the dictionary may be reused
11   * multiple times in the segmentation.
12   *
13   * <pre>
14   *     Example 1:
15   *     Input: s = "microsoft", wordDict = ["micro","soft"]
16   *     Output: true
17   *
18   *     Example 2:
19   *     Input: s = "appletenapple", wordDict = ["apple","ten"]
20   *     Output: true
21   *     Note that you are allowed to reuse a dictionary word.
22   *
23   *     Example 3:
24   *     Input: s = "catsandog", wordDict = ["cats","dog","sand","and","cat"]
25   *     Output: false
26   * </pre>
27   * Constraints -
28   * <pre>
29   *     1 &le; s.length &le; 300
30   *     1 &le; wordDict.length &le; 1000
31   *     1 &le; wordDictfil length &le; 20
32   * </pre>
33   * @author ShortThirdMan
34   */
35  public class WordBreaker {
36  
37      public boolean wordBreak(String s, List<String> wordDict) {
38          // Convert the list to a set for O(1) lookups
39          Set<String> wordSet = new HashSet<>(wordDict);
40  
41          // Create a DP array where dp[i] means whether s.substring(0, i) can be segmented
42          boolean[] dp = new boolean[s.length() + 1];
43          dp[0] = true; // An empty string can always be segmented
44  
45          // Iterate over each position in the string
46          for (int i = 1; i <= s.length(); i++) {
47              // Check every possible start position for the current end position i
48              for (int j = 0; j < i; j++) {
49                  // Check only if the prefix up to j can be segmented and
50                  // the substring from j to i is in the dictionary
51                  if (dp[j] && wordSet.contains(s.substring(j, i))) {
52                      dp[i] = true;
53                      break; // No need to check further, we've found a valid segmentation
54                  }
55              }
56          }
57  
58          // The answer is whether the entire string can be segmented
59          return dp[s.length()];
60      }
61  }
