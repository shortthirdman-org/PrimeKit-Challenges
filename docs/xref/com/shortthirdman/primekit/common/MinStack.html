1   package com.shortthirdman.primekit.common;

3   import lombok.Data;
4   
5   import java.util.Stack;
6   import java.util.function.BooleanSupplier;
7   
8   @Data
9   public class MinStack {
10  
11      private Stack<Integer> stack;
12      private int minElement;
13  
14      public MinStack() {
15          stack = new Stack<>();
16          minElement = Integer.MAX_VALUE;
17      }
18  
19      public void push(int val) {
20          if (val <= minElement) {
21              // Push current minimum twice to track the previous minimum later.
22              stack.push(minElement);
23              minElement = val;
24          }
25          stack.push(val);
26      }
27  
28      public void pop() {
29          if (stack.peek() == minElement) {
30              // If popping the min, update minEle to the previously stored minimum.
31              minElement = stack.pop();
32              stack.pop();
33          } else {
34              stack.pop();
35          }
36      }
37  
38      public int top() {
39          return stack.peek();
40      }
41  
42      public int getMin() {
43          return minElement;
44      }
45  
46      public BooleanSupplier isEmpty() {
47          return () -> stack.empty();
48      }
49  }
