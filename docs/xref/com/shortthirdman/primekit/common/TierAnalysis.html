5   public record TierAnalysis(int orderCount, BigDecimal totalRevenue, long totalItems) {
6   
7       public double getAverageOrderValue() {
8           return orderCount == 0 ? 0 : totalRevenue.doubleValue() / orderCount;
9       }
10  
11      public TierAnalysis combine(TierAnalysis other) {
12          return new TierAnalysis(
13                  this.orderCount + other.orderCount,
14                  this.totalRevenue.add(other.totalRevenue),
15                  this.totalItems + other.totalItems
16          );
17      }
18  
19      public TierAnalysis combine(TierAnalysis a, TierAnalysis b) {
20          return new TierAnalysis(
21                  a.orderCount + b.orderCount,
22                  a.totalRevenue.add(b.totalRevenue),
23                  a.totalItems + b.totalItems
24          );
25      }
26  }
