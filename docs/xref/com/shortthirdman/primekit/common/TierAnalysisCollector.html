3   import java.math.BigDecimal;
4   import java.util.Collections;
5   import java.util.EnumSet;
6   import java.util.Set;
7   import java.util.function.BiConsumer;
8   import java.util.function.BinaryOperator;
9   import java.util.function.Function;
10  import java.util.function.Supplier;
11  import java.util.stream.Collector;
12  
13  /**
14   * Custom collector to analyze orders by customer tier, showing average order value and total items.
15   * @author ShortThirdMan
16   * @since 1.1.1
17   */
18  public class TierAnalysisCollector implements Collector<Order, TierAnalysis, TierAnalysis> {
19  
20      @Override
21      public Supplier<TierAnalysis> supplier() {
22          return () -> new TierAnalysis(0, BigDecimal.ZERO, 0);
23      }
24  
25      @Override
26      public BiConsumer<TierAnalysis, Order> accumulator() {
27          return (analysis, order) -> {
28              BigDecimal orderTotal = order.items().stream()
29                      .map(item -> item.product().price()
30                              .multiply(BigDecimal.valueOf(item.quantity())))
31                      .reduce(BigDecimal.ZERO, BigDecimal::add);
32  
33              long itemCount = order.items().stream()
34                      .mapToLong(OrderItem::quantity)
35                      .sum();
36  
37              analysis = new TierAnalysis(
38                      analysis.orderCount() + 1,
39                      analysis.totalRevenue().add(orderTotal),
40                      analysis.totalItems() + itemCount
41              );
42          };
43      }
44  
45      @Override
46      public BinaryOperator<TierAnalysis> combiner() {
47          return TierAnalysis::combine;
48      }
49  
50      @Override
51      public Function<TierAnalysis, TierAnalysis> finisher() {
52          return Function.identity();
53      }
54  
55      @Override
56      public Set<Characteristics> characteristics() {
57          return Collections.unmodifiableSet(EnumSet.of(
58                  Characteristics.IDENTITY_FINISH
59          ));
60      }
61  }
