10   * Utility class providing algorithms for solving common parentheses-related problems.
11   *
12   * @author ShortThirdMan
13   * @since 1.0.1
14   */
15  public class ParenthesesOps {
16  
17      /**
18       * Returns the length of the longest valid (well-formed) parentheses substring
19       * within the given string. A valid substring consists of properly matched
20       * opening '(' and closing ')' parentheses.
21       *
22       * <p>Examples:
23       * <ul>
24       *   <li>Input: "(()" → Output: 2 ("()")</li>
25       *   <li>Input: ")()())" → Output: 4 ("()()")</li>
26       *   <li>Input: "" → Output: 0</li>
27       * </ul>
28       *
29       * <p>The algorithm runs in O(n) time using a stack to track indices of unmatched
30       * parentheses. Space complexity is O(n) in the worst case.
31       *
32       * @param s the input string containing only characters '(' and ')'.
33       * @return the length of the longest valid parentheses substring. Returns 0 if
34       *         the input is null, empty, or contains no valid substring.
35       * @throws IllegalArgumentException if the input length exceeds 3 * 10^4 or if
36       *         the string contains characters other than '(' or ')'.
37       */
38      public int longestValidParentheses(String s) {
39          if (s == null || s.isEmpty()) {
40              return 0; // constraint check
41          }
42  
43          if (s.length() > 30000) {
44              throw new IllegalArgumentException("Input exceeds max length of 3 * 10^4");
45          }
46  
47          int maxLen = 0;
48          Deque<Integer> stack = new ArrayDeque<>();
49          stack.push(-1);
50  
51          for (int i = 0; i < s.length(); i++) {
52              char c = s.charAt(i);
53  
54              switch (c) {
55                  case '(':
56                      stack.push(i);
57                      break;
58  
59                  case ')':
60                      stack.pop();
61                      if (stack.isEmpty()) {
62                          stack.push(i);
63                      } else {
64                          maxLen = Math.max(maxLen, i - stack.peek());
65                      }
66                      break;
67  
68                  default:
69                      // If constraints are trusted, this should never occur.
70                      throw new IllegalArgumentException("Invalid character: " + c);
71              }
72          }
73  
74          return maxLen;
75      }

77      /**
78       * Generates all combinations of well-formed parentheses for a given number of pairs.
79       *
80       * <p>A well-formed parentheses string must satisfy:
81       * <ul>
82       *   <li>Every opening '(' has a corresponding closing ')'.</li>
83       *   <li>No prefix of the string has more ')' than '('.</li>
84       * </ul>
85       *
86       * <p>Examples:
87       * <ul>
88       *   <li>Input: n = 3 → Output: ["((()))","(()())","(())()","()(())","()()()"]</li>
89       *   <li>Input: n = 1 → Output: ["()"]</li>
90       * </ul>
91       *
92       * @param n the number of pairs of parentheses (1 ≤ n ≤ 8).
93       * @return a list of all well-formed parentheses combinations.
94       * @throws IllegalArgumentException if n is outside the range [1, 8].
95       */
96      public List<String> generateParenthesis(int n) {
97          if (n < 1 || n > 8) {
98              throw new IllegalArgumentException("n must be between 1 and 8");
99          }
100 
101         List<String> result = new ArrayList<>();
102         // Define a recursive lambda (Java 8 trick: self-reference via array wrapper)
103         BiConsumer<StringBuilder, int[]> dfs = new BiConsumer<>() {
104             @Override
105             public void accept(StringBuilder current, int[] state) {
106                 int open = state[0], close = state[1];
107 
108                 if (current.length() == n * 2) {
109                     result.add(current.toString());
110                     return;
111                 }
112 
113                 if (open < n) {
114                     current.append('(');
115                     accept(current, new int[]{open + 1, close});
116                     current.deleteCharAt(current.length() - 1);
117                 }
118                 if (close < open) {
119                     current.append(')');
120                     accept(current, new int[]{open, close + 1});
121                     current.deleteCharAt(current.length() - 1);
122                 }
123             }
124         };
125 
126         dfs.accept(new StringBuilder(), new int[]{0, 0});
127 
128         return result;
129     }
