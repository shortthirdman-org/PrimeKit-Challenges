1   package com.shortthirdman.primekit.hackerearth;
2   
3   import java.util.HashMap;
4   import java.util.List;
5   import java.util.Map;
6   import java.util.Objects;
7   
8   /**
9    * Manages the lifecycle, renewal, and state evaluation of authentication tokens.
10   * <p>
11   * This class tracks token creation and reset commands over discrete time units,
12   * enforcing global expiration limits and determining token availability at the
13   * conclusion of event processing.
14   * </p>
15   *
16   * @author ShortThirdMan
17   * @since 1.2.1
18   */
19  public class TokenExpiryTracker {
20  
21      private static final int CMD_TYPE_INDEX = 0;
22      private static final int CMD_TOKEN_ID_INDEX = 1;
23      private static final int CMD_TIME_INDEX = 2;
24  
25      private static final int ACTION_CREATE = 0;
26      private static final int ACTION_RESET = 1;
27  
28      /**
29       * Calculates the number of tokens that remain active after executing all specified commands.
30       * <p>
31       * A token is active immediately upon creation at time {@code T} and expires at {@code T + expiryLimit}.
32       * Subsequent reset commands at time {@code T} are valid only if the token already exists and has not yet
33       * expired (i.e., its current expiration time is greater than or equal to {@code T}).
34       * </p>
35       * <p>
36       * After processing all commands, a token is considered active if its expiration time is strictly
37       * greater than the timestamp of the last executed command.
38       * </p>
39       *
40       * @param expiryLimit the fixed lifespan added to the execution timestamp upon token creation or reset;
41       *                    must be greater than or equal to 1
42       * @param commands    a chronological list of 3-element integer tuples representing operations:
43       *                    <ul>
44       *                      <li>{@code [0, tokenId, T]} - Creates a new token with ID {@code tokenId} at time {@code T}.</li>
45       *                      <li>{@code [1, tokenId, T]} - Resets the expiration of an existing token with ID {@code tokenId} at time {@code T}.</li>
46       *                    </ul>
47       *                    Must be sorted in non-decreasing order of {@code T}.
48       * @return the total count of active tokens remaining after all commands have been evaluated;
49       *         returns {@code 0} if {@code commands} is empty
50       * @throws NullPointerException     if {@code commands} is {@code null} or contains {@code null} entries
51       * @throws IllegalArgumentException if {@code expiryLimit} is non-positive or if any command does not
52       *                                  contain exactly three elements
53       */
54      public int numberOfTokens(int expiryLimit, List<List<Integer>> commands) {
55          Objects.requireNonNull(commands, "commands list must not be null");
56          if (expiryLimit <= 0) {
57              throw new IllegalArgumentException("expiryLimit must be positive: " + expiryLimit);
58          }
59          if (commands.isEmpty()) {
60              return 0;
61          }
62  
63          Map<Integer, Integer> tokenExpiries = new HashMap<>();
64  
65          for (List<Integer> cmd : commands) {
66              Objects.requireNonNull(cmd, "command entry must not be null");
67              if (cmd.size() != 3) {
68                  throw new IllegalArgumentException("Command must contain exactly 3 integers: [type, tokenId, T]");
69              }
70  
71              int type = cmd.get(CMD_TYPE_INDEX);
72              int tokenId = cmd.get(CMD_TOKEN_ID_INDEX);
73              int timestamp = cmd.get(CMD_TIME_INDEX);
74  
75              if (type == ACTION_CREATE) {
76                  tokenExpiries.put(tokenId, timestamp + expiryLimit);
77              } else if (type == ACTION_RESET) {
78                  Integer currentExpiry = tokenExpiries.get(tokenId);
79                  if (currentExpiry != null && currentExpiry >= timestamp) {
80                      tokenExpiries.put(tokenId, timestamp + expiryLimit);
81                  }
82              }
83          }
84  
85          int finalTimestamp = commands.getLast().get(CMD_TIME_INDEX);
86  
87          return (int) tokenExpiries.values().stream()
88                  .filter(expiry -> expiry > finalTimestamp)
89                  .count();
90      }
91  }
