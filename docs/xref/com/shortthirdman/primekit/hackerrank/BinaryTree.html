19      /**
20       * Given a binary tree, find its minimum depth.
21       * <br></br>
22       * The minimum depth is the number of nodes along the shortest path from the root node down to the nearest leaf node.
23       * A leaf is a node with no children.
24       *
25       * @param root the node with its leaves
26       * @return the minimum depth
27       * @see <a href="https://leetcode.com/problems/minimum-depth-of-binary-tree/">Minimum Depth of Binary Tree</a>
28       */
29      public int minDepth(TreeNode root) {
30  
31          if (root == null) {
32              return 0; // Base case: if the tree is empty
33          }
34  
35          // If the node is a leaf node
36          if (root.getLeft() == null && root.getRight() == null) {
37              return 1; // Count this node
38          }
39  
40          // If the left child is null, we only consider the right child
41          if (root.getLeft() == null) {
42              return 1 + minDepth(root.getRight());
43          }
44  
45          // If the right child is null, we only consider the left child
46          if (root.getRight() == null) {
47              return 1 + minDepth(root.getLeft());
48          }
49  
50          // If both children are present, return the minimum depth of both subtrees
51          return 1 + Math.min(minDepth(root.getLeft()), minDepth(root.getRight()));
52      }

54      /**
55       * Given the root of a binary tree, return its maximum depth.
56       * <br></br>
57       * A binary tree's maximum depth is the number of nodes along the longest path from the root node down to the farthest leaf node.
58       *
59       * @param root the node with its leaves
60       * @return the maximum depth
61       * @see <a href="https://leetcode.com/problems/minimum-depth-of-binary-tree/">Minimum Depth of Binary Tree</a>
62       */
63      public int maxDepth(TreeNode root) {
64          if (root == null) {
65              return 0;
66          }
67  
68          // Calculate depth recursively for left and right subtrees
69          return Math.max(maxDepth(root.left), maxDepth(root.right)) + 1;
70      }

72      /**
73       * Given a binary tree, determine if it is height-balanced.
74       *
75       * @param root the node with its leaves
76       * @return true if height is balanced, false otherwise
77       * @see <a href="https://leetcode.com/problems/balanced-binary-tree/description/">Balanced Binary Tree</a>
78       */
79      public boolean isBalanced(TreeNode root) {
80          if (root == null) {
81              return true; // An empty tree is balanced
82          }
83  
84          // Stack to simulate recursion
85          Deque<TreeNode> stack = new ArrayDeque<>();
86          // Map to store the height of each node
87          Map<TreeNode, Integer> heights = new HashMap<>();
88  
89          stack.push(root);
90  
91          while (!stack.isEmpty()) {
92              TreeNode node = stack.peek();
93  
94              // If the node has not been processed yet, push its children onto the stack
95              if (!heights.containsKey(node)) {
96                  if (node.right != null) {
97                      stack.push(node.right);
98                  }
99                  if (node.left != null) {
100                     stack.push(node.left);
101                 }
102                 // Mark the node as processed by adding it to the heights map
103                 heights.put(node, 0);
104             } else {
105                 // If the node has been processed, calculate its height
106                 stack.pop(); // Remove the node from the stack
107 
108                 int leftHeight = heights.getOrDefault(node.left, 0);
109                 int rightHeight = heights.getOrDefault(node.right, 0);
110 
111                 // Check if the current node is balanced
112                 if (Math.abs(leftHeight - rightHeight) > 1) {
113                     return false; // Tree is unbalanced
114                 }
115 
116                 // Calculate the height of the current node
117                 int height = Math.max(leftHeight, rightHeight) + 1;
118                 heights.put(node, height);
119             }
120         }
121 
122         return true; // If all nodes are balanced
123     }

125     /**
126      * @param root the root tree node
127      * @return the depth of the tree
128      */
129     public int minimumDepth(TreeNode root) {
130         if (root == null) return 0;
131 
132         Queue<TreeNode> queue = new LinkedList<>();
133         queue.add(root);
134         int depth = 1; // Start with depth 1 since we start from the root
135 
136         while (!queue.isEmpty()) {
137             int levelSize = queue.size();
138 
139             for (int i = 0; i < levelSize; i++) {
140                 TreeNode node = queue.poll();
141 
142                 // Check for null node (just in case)
143                 if (node == null) continue;
144 
145                 // Check if we have reached a leaf node
146                 if (node.getLeft() == null && node.getRight() == null) {
147                     return depth; // Return the current depth when we find the first leaf
148                 }
149 
150                 // Add children to the queue
151                 if (node.left != null) {
152                     queue.add(node.left);
153                 }
154                 if (node.right != null) {
155                     queue.add(node.right);
156                 }
157             }
158             depth++; // Increase the depth for the next level
159         }
160 
161         return depth; // In case of an empty tree, though not reached due to checks
162     }

164     /**
165      * Given the root of a binary tree, check whether it is a mirror of itself (i.e., symmetric around its center).
166      *
167      * @param root the node with its leaves
168      * @return true if tree is symmetric, otherwise false
169      * @see <a href="https://leetcode.com/problems/symmetric-tree/description/">Symmetric Tree</a>
170      */
171     public boolean isSymmetric(TreeNode root) {
172         if (root == null) {
173             return true; // An empty tree is symmetric
174         }
175 
176         Stack<TreeNode> stack = new Stack<>();
177         stack.push(root.left);
178         stack.push(root.right);
179 
180         while (!stack.isEmpty()) {
181             TreeNode right = stack.pop();
182             TreeNode left = stack.pop();
183 
184             // Both nodes are null, continue checking
185             if (left == null && right == null) {
186                 continue;
187             }
188 
189             // If one is null or their values don't match, it's not symmetric
190             if (left == null || right == null || left.value != right.value) {
191                 return false;
192             }
193 
194             // Push children in the order that ensures symmetry
195             stack.push(left.left);
196             stack.push(right.right);
197             stack.push(left.right);
198             stack.push(right.left);
199         }
200         return true; // If we finish checking without issues, it's symmetric
201     }
