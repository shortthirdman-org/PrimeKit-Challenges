3   import java.util.Arrays;
4   import java.util.stream.IntStream;
5   
6   /**
7    * A team of data analysts at Amazon is working to identify data patterns.
8    * During their analysis, they found a category of string they call dominant string:
9    * <br><br/>
10   * - It has an even length.
11   * <br><br/>
12   * - The string contains at least one character with a frequency that matches exactly half of the length of the string.
13   * <br><br/>
14   * Given a string <code>s</code> of length <code>n</code>, determine the number of its distinct substrings that are dominant strings.
15   *
16   * @author ShortThirdMan
17   */
18  public class DominantStringAnalyzer {
19  
20      /**
21       * @param s the input text
22       * @return the count or number of dominant string
23       */
24      public long getDominantStringCount(String s) {
25          long count = 0;
26          int n = s.length();
27          var lowercaseText = s.toLowerCase();
28  
29          count = IntStream.range(2, n + 1)
30                  .filter(len -> len % 2 == 0)  // Only consider even lengths
31                  .mapToLong(len ->
32                          IntStream.range(0, n - len + 1)
33                                  .mapToObj(i -> lowercaseText.substring(i, i + len))
34                                  .filter(substring -> isDominant(substring, len))
35                                  .count()
36                  ).sum();
37  
38          return count;
39      }
40  
41      private boolean isDominant(String substring, int len) {
42          int[] frequency = new int[10];
43          substring.chars()
44                  .forEach(c -> {
45                      // Ensure the character is between 'a' and 'z' before updating frequency
46                      if (c >= 'a' && c <= 'z') {
47                          frequency[c - 'a']++;
48                      }
49                  });
50  
51          int halfLength = len / 2;
52          return Arrays.stream(frequency)
53                  .anyMatch(f -> f == halfLength);
54      }
55  }
