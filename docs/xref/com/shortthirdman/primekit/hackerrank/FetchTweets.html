42      public List<Integer> fetchTweets(int xCoordOfUser, int yCoordOfUser, int K) {
43          if (K < 1 || K > 20) {
44              throw new IllegalArgumentException("K should be in the range [1, 20]");
45          }
46  
47          if (xCoordOfUser < 1 || yCoordOfUser < 1 || xCoordOfUser > 10 || yCoordOfUser > 10) {
48              throw new IllegalArgumentException("Coordinate values for X and Y should be in the range [1, 10]");
49          }
50  
51          // Priority queue to store tweets sorted by distance
52          PriorityQueue<Tweet> pq = new PriorityQueue<>((t1, t2) -> {
53              // Calculate squared Euclidean distance to avoid floating-point operations
54              int x1Squared = (int) Math.pow(t1.getXCoord() - xCoordOfUser, 2);
55              int y1Squared = (int) Math.pow(t1.getYCoord() - yCoordOfUser, 2);
56              int dist1 = x1Squared + y1Squared;
57  
58              int x2Squared = (int) Math.pow(t2.getXCoord() - xCoordOfUser, 2);
59              int y2Squared = (int) Math.pow(t2.getYCoord() - yCoordOfUser, 2);
60              int dist2 = x2Squared + y2Squared;
61  
62              return Integer.compare(dist1, dist2);
63          });
64  
65          // Add all tweets to the priority queue
66          List<Integer> result = new ArrayList<>();
67          if (tweets.isEmpty()) {
68              return result;
69          }
70          pq.addAll(tweets);
71  
72          // Fetch the K-closest tweets
73          while (K > 0 && !pq.isEmpty()) {
74              result.add(pq.poll().getId());
75              K--;
76          }
77  
78          return result;
79      }

81      public List<Integer> fetchClosestTweets(int xCoordOfUser, int yCoordOfUser, int K) {
82          if (K < 1 || K > 20) {
83              throw new IllegalArgumentException("K should be in the range [1, 20]");
84          }
85  
86          if (xCoordOfUser < 1 || yCoordOfUser < 1 || xCoordOfUser > 10 || yCoordOfUser > 10) {
87              throw new IllegalArgumentException("Coordinate values for X and Y should be in the range [1, 10]");
88          }
89  
90          List<Integer> closestTweets = new ArrayList<>();
91          tweets.sort(Comparator.comparingInt(t -> Math.abs(t.getXCoord() - xCoordOfUser) + Math.abs(t.getYCoord() - yCoordOfUser)));
92  
93          if (tweets.isEmpty()) {
94              return closestTweets;
95          }
96  
97          for (int i = 0; i < K && i < tweets.size(); i++) {
98              closestTweets.add(tweets.get(i).getId());
99          }
100 
101         return closestTweets;
102     }
