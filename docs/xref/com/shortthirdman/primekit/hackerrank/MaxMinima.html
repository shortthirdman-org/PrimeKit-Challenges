46      public int maxMin(List<Integer> arr, int k) {
47          int n = arr.size();
48          int log = (int) (Math.log(n) / Math.log(2));
49  
50          if (k < 1 || k > n) {
51              throw new IllegalArgumentException("k should be between 1 and input array size");
52          }
53  
54          boolean elementGrTh1 = arr.stream().anyMatch(e -> e == 0);
55          if (elementGrTh1) {
56              throw new IllegalArgumentException("Element should be between 1 and 10^9");
57          }
58  
59          // Sparse Table
60          int[][] st = new int[n][log + 1];
61  
62          // Initialize level 0 with the original array
63          for (int i = 0; i < n; i++) {
64              st[i][0] = arr.get(i);
65          }
66  
67          // Build Sparse Table
68          for (int j = 1; j <= log; j++) {
69              int len = 1 << j;
70              int half = len >> 1;
71              for (int i = 0; i + len <= n; i++) {
72                  st[i][j] = Math.min(st[i][j - 1], st[i + half][j - 1]);
73              }
74          }
75  
76          // Query each window of size k
77          int maxOfMins = Integer.MIN_VALUE;
78          int p = (int) (Math.log(k) / Math.log(2));
79          int powLen = 1 << p;
80  
81          for (int i = 0; i + k <= n; i++) {
82              int left = st[i][p];
83              int right = st[i + k - powLen][p];
84              int windowMin = Math.min(left, right);
85  
86              maxOfMins = Math.max(maxOfMins, windowMin);
87          }
88  
89          return maxOfMins;
90      }
