3   /**
4    * There are n people standing in a line labeled from <code>1 to n</code>. The first person in the line is holding a pillow initially.
5    * Every second, the person holding the pillow passes it to the next person standing in the line. Once the pillow reaches the end of the line,
6    * the direction changes, and people continue passing the pillow in the opposite direction.
7    * <br></br>
8    * For example, once the pillow reaches the <code>nth</code> person they pass it to the <code>n - 1th</code> person,
9    * then to the <code>n - 2th</code> person and so on.
10   * Given the two positive integers <code>n</code> and <code>time</code>, return the <b>index</b> of the person holding the pillow
11   * after <b>time</b> seconds.
12   *
13   * @author ShortThirdMan
14   * @see <a href="https://leetcode.com/problems/pass-the-pillow/description/">LeetCode</a>
15   */
16  public class PassThePillow {
17  
18      public int pillowPass(int n, int time) {
19          if (n < 2 || n > 1000 || time < 1 || time > 1000) {
20              throw new IllegalArgumentException("Invalid input for n or time");
21          }
22  
23          int cycleTime = 2 * (n - 1);
24          int effectiveTime = time % cycleTime;
25  
26          if (effectiveTime <= n - 1) {
27              return 1 + effectiveTime;
28          } else {
29              return n - (effectiveTime - (n - 1));
30          }
31      }
32  }
