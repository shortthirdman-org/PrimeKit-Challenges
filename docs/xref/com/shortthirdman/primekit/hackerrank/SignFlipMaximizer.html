3   import java.util.List;

5   /**
6    * Given an array <code>arr</code> of size <code>n</code> and an integer <code>k</code> representing the number of sign-flip
7    * operations to perform.
8    * A sign-flip operation changes the sign of a single element in the array (from positive to negative or vice versa).
9    * Implement a function that calculates the maximum possible sum of the array after performing exactly k sign-flip operations.
10   *
11   * @author shortthirdman
12   */
13  public class SignFlipMaximizer {
14  
15      /**
16       * @param arr the input array
17       * @param k the total number of sign-flip operations performed
18       * @return the maximum possible sum of the array after performing exactly <code>k</code> sign-flip operations
19       */
20      public long maximizeSum(List<Integer> arr, int k) {
21          // Step 1: Sort the array by absolute value in descending order
22          arr.sort((a, b) -> Integer.compare(Math.abs(b), Math.abs(a)));
23  
24          // Step 2: Flip as many negative numbers as possible
25          for (int i = 0; i < arr.size() && k > 0; i++) {
26              if (arr.get(i) < 0) {
27                  arr.set(i, -arr.get(i));
28                  k--;
29              }
30          }
31  
32          // Step 3: If k is still odd, flip the smallest (absolute) value
33          if (k % 2 != 0) {
34              // Find the smallest absolute value element and flip it
35              int minAbsIndex = 0;
36              for (int i = 1; i < arr.size(); i++) {
37                  if (Math.abs(arr.get(i)) < Math.abs(arr.get(minAbsIndex))) {
38                      minAbsIndex = i;
39                  }
40              }
41              arr.set(minAbsIndex, -arr.get(minAbsIndex));
42          }
43  
44          // Step 4: Calculate the final sum
45          long totalSum = 0;
46          for (int num : arr) {
47              totalSum += num;
48          }
49  
50          return totalSum;
51      }
52  }
