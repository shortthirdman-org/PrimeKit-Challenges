4   import java.util.Arrays;
5   import java.util.HashSet;
6   import java.util.List;

8   import java.util.stream.Collectors;

10  /**
11   * Spam Text Classifier:
12   *   A text is "spam" if it contains ≥ 2 occurrences of words from spamWords (case-insensitive, exact matches). Otherwise, return "not_spam".
13   * @author ShortThirdMan
14   * @since 1.1.2
15   */
16  public class SpamTextClassifier {
17  
18      /**
19       * Classifies a list of texts as "SPAM" or "NOT SPAM" based on the presence of spam words.
20       * @param texts the array of texts to classify as spam or not spam
21       * @param spamWords an array of words that are considered spam indicators
22       * @return a list of strings, where each element is "SPAM" if the corresponding text contains any of the spam words, and "NOT SPAM" otherwise
23       */
24      public List<String> classifyTexts(List<String> texts, List<String> spamWords) {
25          if (texts == null || spamWords == null) {
26              throw new NullPointerException("texts and spam words cannot be null");
27          }
28  
29          if (texts.isEmpty()) {
30              return List.of();
31          }
32  
33          Set<String> spamSet = new HashSet<>();
34          for (String word : spamWords) {
35              spamSet.add(word.toLowerCase());
36          }
37  
38          List<String> result = new ArrayList<>(texts.size());
39  
40          for (String text : texts) {
41  
42              int spamCount = 0;
43  
44              // Split words by a single space
45              String[] words = text.split(" ");
46  
47              for (String word : words) {
48                  if (spamSet.contains(word.toLowerCase())) {
49                      spamCount++;
50  
51                      // Early exit optimization
52                      if (spamCount >= 2) {
53                          result.add("spam");
54                          break;
55                      }
56                  }
57              }
58  
59              if (spamCount < 2) {
60                  result.add("not_spam");
61              }
62          }
63  
64          return result;
65      }
66  
67      /**
68       * Classifies a list of texts as "SPAM" or "NOT SPAM" based on the presence of spam words.
69       * @param texts the list of texts to classify
70       * @param spamWords the list of spam words to check against
71       * @return a list of classifications ("spam" or "not_spam") for each text in the input list,
72       * processed in parallel for improved performance
73       */
74      public List<String> classifyTextStream(List<String> texts, List<String> spamWords) {
75          if (texts == null || spamWords == null) {
76              throw new NullPointerException("texts and spam words cannot be null");
77          }
78  
79          if (texts.isEmpty()) {
80              return List.of();
81          }
82  
83          Set<String> spamSet = spamWords.stream()
84                  .map(String::toLowerCase)
85                  .collect(Collectors.toSet());
86  
87          return texts.parallelStream()
88                  .map(text -> {
89                      long spamCount = Arrays.stream(text.split(" "))
90                              .map(String::toLowerCase)
91                              .filter(spamSet::contains)
92                              .limit(2)
93                              .count();
94  
95                      return spamCount >= 2 ? "spam" : "not_spam";
96                  })
97                  .toList();
98      }
99  }
