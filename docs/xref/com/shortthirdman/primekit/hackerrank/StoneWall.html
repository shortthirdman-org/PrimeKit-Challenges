3   /**
4    * You are going to build a stone wall. The wall should be straight and <code>N</code> meters long, and its thickness should be constant;
5    * however, it should have different heights in different places. The height of the wall is specified by an array <code>H</code> of N positive integers.
6    * <code>H[i]</code> is the height of the wall from <code>i to i+1</code> meters to the right of its left end.
7    * <br>
8    * In particular, <code>H[0]</code> is the height of the wall's left end and <code>H[N−1]</code> is the height of the wall's right end.
9    * The wall should be built of cuboid stone blocks (that is, all sides of such blocks are rectangular).
10   *
11   * @author ShortThirdMan
12   */
13  public class StoneWall {
14  
15      /**
16       * Compute the minimum number of blocks needed to build the wall.
17       * @param H array of positive integers specifying height of wall
18       * @return Minimum number of blocks needed to build
19       */
20      public int coverManhattanSkyline(int[] H) {
21          int N = H.length;
22          int totalBlocks = 0;
23  
24          if (N < 1 || N > 100000) {
25              throw new IllegalArgumentException("Number of height-blocks should be in range [1,100000]");
26          }
27  
28          for (int height : H) {
29              if (height <= 0) {
30                  throw new IllegalArgumentException("Height values must be positive");
31              }
32              totalBlocks += height;
33          }
34  
35          return totalBlocks;
36      }
37  }
