3   import java.util.Arrays;
4   
5   /**
6    * There are <code>n</code> students in a class numbered from <code>0</code> to <code>n - 1</code>.
7    * The teacher will give each student a problem starting with the student number <code>0</code>, then the student number <code>1</code>, and so on
8    * until the teacher reaches the student number <code>n - 1</code>. After that, the teacher will restart the process,
9    * starting with the student number <code>0</code> again.<br><br/>
10   *
11   * You are given a <b>0-indexed</b> integer array chalk and an integer <code>k</code>.
12   * There are initially <code>k</code> pieces of chalk. When the student number <code>i</code> is given a problem to solve,
13   * they will use <code>chalk[i]</code> pieces of chalk to solve that problem.
14   * However, if the current number of chalk pieces is strictly less than <code>chalk[i]</code>, then the student number <code>i</code>
15   * will be asked to replace the chalk. <br><br/>
16   *
17   * Return the <b>index</b> of the student that will replace the chalk pieces.
18   * @author ShortThirdMan
19   * @see <a href="https://leetcode.com/problems/find-the-student-that-will-replace-the-chalk/">LeetCode</a>
20   */
21  public class StudentChalkReplacer {
22  
23      public int chalkReplacer(int[] chalk, int k) {
24          if (chalk == null) {
25              throw new NullPointerException("Chalk can not be null");
26          }
27  
28          if (chalk.length == 0) {
29              throw new IllegalArgumentException("Number of chalks cannot be zero");
30          }
31  
32          boolean allPositive = Arrays.stream(chalk).allMatch(n -> n > 0);
33          if (!allPositive) {
34              throw new IllegalArgumentException("Values for chalk can not be negative");
35          }
36  
37          if (k < 0) {
38              throw new IllegalArgumentException("k can not be negative");
39          }
40  
41          // Calculate the total chalk needed for one round
42          int totalChalk = 0;
43          for (int i = 0; i < chalk.length; i++) {
44              totalChalk += chalk[i];
45          }
46  
47          // Reduce k using the modulo operation
48          k %= totalChalk;
49  
50          // Find the student who will replace the chalk
51          for (int i = 0; i < chalk.length; i++) {
52              if (k < chalk[i]) {
53                  return i;
54              }
55              k -= chalk[i];
56          }
57  
58          // Should never reach here if input constraints are respected
59          return -1;
60      }
61  }
