3   import java.util.Collections;
4   import java.util.List;
5   
6   public class TeamEfficiencyCalculator {
7   
8       public long getTotalEfficiency(List<Integer> skill) {
9           if (skill == null || skill.isEmpty() || skill.size() % 2 != 0) {
10              return -1; // Can't form teams if the list is null or the number of participants isn't even
11          }
12  
13          Collections.sort(skill); // Sort the skills to easily form pairs
14          int n = skill.size();
15  
16          // Additional check for mixed positive and negative skills
17          // This ensures that if there are both negative and positive values, pairing is invalid.
18          if (skill.getFirst() < 0 && skill.get(n - 1) > 0) {
19              return -1;
20          }
21  
22          long totalEfficiency = 0;
23          int targetSum = skill.getFirst() + skill.get(n - 1); // Calculate the target sum of skills for all pairs
24  
25          for (int i = 0; i < n / 2; i++) {
26              int sum = skill.get(i) + skill.get(n - 1 - i); // Sum of skills for the current pair
27              if (sum != targetSum) {
28                  return -1; // Return -1 if a pair doesn't satisfy the condition
29              }
30              totalEfficiency += (long) skill.get(i) * skill.get(n - 1 - i); // Add the product of skills to total efficiency
31          }
32  
33          return totalEfficiency;
34      }
35  }
