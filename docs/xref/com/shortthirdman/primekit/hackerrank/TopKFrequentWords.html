9   /**
10   * <h3>K Most Frequent Words</h3>
11   * Given a list of words, return the k most frequent words sorted by frequency and then lexicographically.
12   * @author ShortThirdMan
13   */
14  public class TopKFrequentWords {
15  
16      public List<String> topKFrequent(String[] words, int k) {
17          // Count frequency for each word.
18          Map<String, Integer> frequencyMap = new HashMap<>();
19          for (String word : words) {
20              frequencyMap.put(word, frequencyMap.getOrDefault(word, 0) + 1);
21          }
22  
23          // Sort words by frequency (higher first) then lexicographically (ascending).
24          List<String> candidates = new ArrayList<>(frequencyMap.keySet());
25          Collections.sort(candidates, (w1, w2) -> {
26              int freqCompare = frequencyMap.get(w2) - frequencyMap.get(w1);
27              if (freqCompare == 0) {
28                  return w1.compareTo(w2);
29              }
30              return freqCompare;
31          });
32  
33          // Return the top k frequent words.
34          return candidates.subList(0, Math.min(k, candidates.size()));
35      }
36  
37      /**
38       * Optimized a top-k frequent collection of word
39       * @param words the input array of words
40       * @param k the most frequent words value
41       * @return
42       */
43      public List<String> optimizeTopKFrequent(String[] words, int k) {
44          if (k <= 0 || words == null || words.length == 0) {
45              return Collections.emptyList();
46          }
47  
48          // Count frequency for each word, including null
49          final Map<String, Integer> frequencyMap = new HashMap<>();
50          for (String word : words) {
51              frequencyMap.put(word, frequencyMap.getOrDefault(word, 0) + 1);
52          }
53  
54          // Sort words by frequency (higher first) then lexicographically (null-safe)
55          List<String> candidates = new ArrayList<>(frequencyMap.keySet());
56          candidates.sort((w1, w2) -> {
57              int freqCompare = frequencyMap.get(w2) - frequencyMap.get(w1);
58              if (freqCompare == 0) {
59                  // null-safe lexicographic ordering
60                  if (w1 == null && w2 == null) return 0;
61                  if (w1 == null) return 1;  // put nulls last
62                  if (w2 == null) return -1; // put nulls last
63                  return w1.compareTo(w2);
64              }
65              return freqCompare;
66          });
67  
68          // Return top k frequent words
69          return candidates.subList(0, Math.min(k, candidates.size()));
70      }
