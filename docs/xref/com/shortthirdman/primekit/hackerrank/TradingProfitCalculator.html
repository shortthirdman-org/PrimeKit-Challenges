1   package com.shortthirdman.primekit.hackerrank;
2   
3   import java.util.*;
4   
5   /**
6    * A quantitative trading firm aims to develop a tool track the net profit/loss of the firm
7    * at any point in time. THis tool processes a list of events , and each event fails into one of the
8    * four categories:
9    * <ul>
10   *     <li>
11   *         BUY stock quantity: Indicates the purchase of <em>quantity</em> shares of <em>stock</em> at the market price.
12   *     </li>
13   *     <li>
14   *         SELL stock quantity: Indicates the sale of <em>quantity</em> shares of <em>stock</em> at the market price.
15   *     </li>
16   *     <li>
17   *         CHANGE stock price: Indicates a change in the market price of <em>stock</em> by <em>price</em> amount,
18   *         which can be positive or negative.
19   *     </li>
20   *     <li>
21   *         QUERY: Requests the net profit/loss from the start of trading until the current time.
22   *     </li>
23   * </ul>
24   * The tool should return a list of numbers corresponding to each QUERY event.
25   *
26   * @author ShortThirdMan
27   */
28  public class TradingProfitCalculator {
29  
30      private static final Integer MAX_QUANTITY = 1000;
31      private static final Integer MAX_PRICE_DELTA = 1000;
32  
33      public List<Long> getNetProfit(List<String> events) {
34          Map<String, Integer> holdings = new HashMap<>();
35          Map<String, Integer> prices = new HashMap<>();
36          Map<String, Long> costBasis = new HashMap<>();
37          List<Long> result = new ArrayList<>();
38          long cash = 0;
39  
40          if (events.isEmpty() || events.size() > 100000) {
41              throw new IllegalArgumentException("Number of events must be between 1 and 10^5.");
42          }
43  
44          for (String event : events) {
45              if (event.length() > 21) {
46                  throw new IllegalArgumentException("Event string exceeds 21 characters.");
47              }
48  
49              String[] parts = event.split(" ");
50              String type = parts[0];
51  
52              switch (type) {
53                  case "BUY": {
54                      if (parts.length != 3) throw new IllegalArgumentException("Invalid BUY format.");
55                      String stock = parts[1];
56                      int qty = Integer.parseInt(parts[2]);
57                      if (qty < 1 || qty >= MAX_QUANTITY) {
58                          throw new IllegalArgumentException("BUY quantity out of bounds.");
59                      }
60  
61                      int price = prices.getOrDefault(stock, 0);
62                      holdings.put(stock, holdings.getOrDefault(stock, 0) + qty);
63                      costBasis.put(stock, costBasis.getOrDefault(stock, 0L) + (long) qty * price);
64                      break;
65                  }
66  
67                  case "SELL": {
68                      if (parts.length != 3) {
69                          throw new IllegalArgumentException("Invalid SELL format.");
70                      }
71  
72                      String stock = parts[1];
73                      int qty = Integer.parseInt(parts[2]);
74                      if (qty < 1 || qty >= MAX_QUANTITY) {
75                          throw new IllegalArgumentException("SELL quantity out of bounds.");
76                      }
77  
78                      int owned = holdings.getOrDefault(stock, 0);
79                      if (qty > owned) {
80                          throw new IllegalArgumentException("Not enough shares to sell.");
81                      }
82  
83                      int price = prices.getOrDefault(stock, 0);
84                      holdings.put(stock, owned - qty);
85                      costBasis.put(stock, costBasis.get(stock) - (long) qty * price);
86                      cash += (long) qty * price;
87                      break;
88                  }
89  
90                  case "CHANGE": {
91                      if (parts.length != 3) throw new IllegalArgumentException("Invalid CHANGE format.");
92                      String stock = parts[1];
93                      int delta = Integer.parseInt(parts[2]);
94                      if (Math.abs(delta) > MAX_PRICE_DELTA) {
95                          throw new IllegalArgumentException("Price change exceeds allowed delta.");
96                      }
97  
98                      prices.put(stock, prices.getOrDefault(stock, 0) + delta);
99                      break;
100                 }
101 
102                 case "QUERY": {
103                     long totalValue = cash;
104                     for (String stock : holdings.keySet()) {
105                         int qty = holdings.get(stock);
106                         int price = prices.getOrDefault(stock, 0);
107                         totalValue += (long) qty * price;
108                     }
109 
110                     long totalCost = 0;
111                     for (long cost : costBasis.values()) {
112                         totalCost += cost;
113                     }
114 
115                     result.add(totalValue - totalCost);
116                     break;
117                 }
118 
119                 default:
120                     throw new IllegalArgumentException("Unknown event type: " + type);
121             }
122         }
123 
124         return result;
125     }
126 }
