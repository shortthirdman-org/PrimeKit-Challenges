3   /**
4    * Given a string that contains only three characters: <code>{</code>, <code>}</code>, and <code>*</code>.
5    * A <code>*</code> can represent either a <code>{</code>, a <code>}</code>, or an empty string.
6    * @author ShortThirdMan
7    */
8   public class ValidParenthesesExpression {
9   
10      public boolean isValid(String s) {
11          if (s == null) {
12              throw new NullPointerException("Input string can not be null");
13          }
14          int minOpen = 0; // minimum possible open braces
15          int maxOpen = 0; // maximum possible open braces
16  
17          for (char c : s.toCharArray()) {
18              switch (c) {
19                  case '{':
20                      minOpen++;
21                      maxOpen++;
22                      break;
23                  case '}':
24                      minOpen--;
25                      maxOpen--;
26                      break;
27                  case '*':
28                      minOpen--; // treat as '}'
29                      maxOpen++; // treat as '{'
30                      break;
31                  default:
32                      break;
33              }
34  
35              // If maxOpen becomes negative, too many closing braces
36              if (maxOpen < 0) return false;
37  
38              // minOpen can't be negative (we can use * as empty string)
39              if (minOpen < 0) minOpen = 0;
40          }
41  
42          // Valid if we can have 0 open braces at the end
43          return minOpen <= 0;
44      }
45  }
