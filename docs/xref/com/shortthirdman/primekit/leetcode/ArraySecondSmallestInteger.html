6   public class ArraySecondSmallestInteger {

8       public int findUsingArraySort(int[] arr) {
9           Arrays.sort(arr);
10          int smallest = arr[0];
11  
12          for (int i = 1; i < arr.length; i++) {
13              if (arr[i] != smallest) {
14                  return arr[i];
15              }
16          }
17  
18          return -1;
19      }
20  
21      /**
22       * @param arr the input array of integers
23       * @return the second-smallest integer
24       */
25      public int findUsingSinglePassThrough(int[] arr) {
26          int smallest = Integer.MAX_VALUE;
27          int secondSmallest = Integer.MAX_VALUE;
28  
29          for (int num : arr) {
30              if (num < smallest) {
31                  secondSmallest = smallest;
32                  smallest = num;
33              } else if (num < secondSmallest && num != smallest) {
34                  secondSmallest = num;
35              }
36          }
37  
38          if (secondSmallest == Integer.MAX_VALUE) {
39              return -1;
40          } else {
41              return secondSmallest;
42          }
43      }

45      /**
46       * A min-heap is a priority queue where the element with the minimum value always resides at the root.
47       * By default, PriorityQueue implements a min-heap, so the element with the minimum value is at the root.
48       *
49       * @param arr
50       * @return
51       */
52      public int findUsingMinHeap(int[] arr) {
53          if (arr.length < 2) {
54              return -1;
55          }
56  
57          PriorityQueue<Integer> minHeap = new PriorityQueue<>();
58          for (int num : arr) {
59              if (minHeap.isEmpty() || num != minHeap.peek()) {
60                  minHeap.offer(num);
61              }
62          }
63          // all elements were the same if minHeap size is less than 2
64          if (minHeap.size() < 2) {
65              return -1;
66          }
67          minHeap.poll(); // Remove the smallest element
68          return minHeap.peek(); // Second smallest element is at the root
69      }
