3   import java.util.stream.Stream;

5   /**
6    * Find longest sequence of zeros in binary representation of an integer.
7    * <br/><br/>
8    * A binary gap within a positive integer N is any maximal sequence of consecutive zeros
9    * that is surrounded by ones at both ends in the binary representation of N.
10   * 
11   * @author shortthirdman
12   * @version 1.0.0
13   * 
14   */
15  public class BinaryGap {
16  
17      /**
18       * @param N the number to convert to binary
19       * @return the longest sequnece
20       */
21  	public int findLongestSequence(int N) {
22          if (N == 1) {
23              return 0;
24          }
25  
26          // Convert to binary string and trim trailing zeros
27          String binaryString = Integer.toBinaryString(N).replaceAll("0+$", "");
28  
29          // Find the longest gap length using streams
30          return Stream.of(binaryString.split("1+"))
31                      .filter(gap -> !gap.isEmpty())
32                      .mapToInt(String::length)
33                      .max()
34                      .orElse(0);
35      }
36  }
