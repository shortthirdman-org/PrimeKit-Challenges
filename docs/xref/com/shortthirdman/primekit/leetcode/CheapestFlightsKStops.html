1   package com.shortthirdman.primekit.leetcode;

3   import java.util.HashMap;
4   import java.util.Map;
5   import java.util.PriorityQueue;
6   import java.util.Queue;
7   
8   /**
9    * There are <code>n</code> cities connected by <code>m</code> flights.
10   * Each flight starts from city <code>u</code> and arrives at <code>v</code>
11   * with a price <code>w</code>.
12   * <br><br>
13   * Now given all the cities and flights, together with starting city <code>src</code>
14   * and the destination <code>dest</code>,
15   * your task is to find the cheapest price from <code>src</code> to <code>dest</code>
16   * with up to <code>k</code> stops.
17   * If there is no such route, output -1.
18   * 
19   * @author shortthirdman
20   * 
21   */
22  public class CheapestFlightsKStops {
23  
24  	/**
25  	 * @param n number of cities connected
26  	 * @param flights the flight route
27  	 * @param src the source city
28  	 * @param dest the destination city
29  	 * @param K the maximum number of stops
30  	 * @return the cheapest flight
31  	 */
32  	public int findCheapestPrice(int n, int[][] flights, int src, int dest, int K) {
33  		Map<Integer, Map<Integer, Integer>> map = new HashMap<>();
34  
35  		for (int i = 0; i < n; i++) {
36  			map.put(i, new HashMap<>());
37  		}
38  
39  		for (int[] f : flights) {
40  			map.get(f[0]).put(f[1], f[2]);
41  		}
42  
43  		Queue<int[]> priority = new PriorityQueue<>((a, b) -> a[0] - b[0]);
44  		priority.add(new int[] { 0, src, -1 });
45  
46  		while (!priority.isEmpty()) {
47  			int[] ff = priority.poll();
48  			int fare = ff[0], dst = ff[1], stop = ff[2];
49  			if (dst == dest) {
50  				return fare;
51  			}
52  			if (stop < K) {
53  				Map<Integer, Integer> temp = map.getOrDefault(dst, new HashMap<>());
54  				for (int cc : temp.keySet()) {
55  					priority.add(new int[] { fare + temp.get(cc), cc, stop + 1 });
56  				}
57  			}
58  		}
59  		return -1;
60  	}
61  }
