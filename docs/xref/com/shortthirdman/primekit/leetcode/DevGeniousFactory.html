14  	public static int getMaximumDistinctCount(int[] a, int[] b, int k) {
15          Map<Integer, Integer> aCount = countFrequencies(a);
16          Map<Integer, Integer> bCount = countFrequencies(b);
17  
18          int maxDistinctCount = aCount.size();
19  
20          for (int i = 0; i < k; i++) {
21              int minAElement = aCount.entrySet().stream().min(Map.Entry.comparingByValue()).get().getKey();
22              int maxBElement = bCount.entrySet().stream().max(Map.Entry.comparingByValue()).get().getKey();
23  
24              if (bCount.get(maxBElement) > aCount.get(minAElement)) {
25                  aCount.put(minAElement, aCount.get(minAElement) + 1);
26                  aCount.put(maxBElement, aCount.get(maxBElement) - 1);
27                  bCount.put(maxBElement, bCount.get(maxBElement) + 1);
28                  bCount.put(minAElement, bCount.get(minAElement) - 1);
29  
30                  maxDistinctCount = Math.max(maxDistinctCount, aCount.size());
31              }
32          }
33  
34          return maxDistinctCount;
35      }

37  	private static Map<Integer, Integer> countFrequencies(int[] arr) {
38          return Arrays.stream(arr).boxed()
39                  .collect(Collectors.toMap(e -> e, e -> 1, Integer::sum));
40      }

91  	public static Character firstNonRepeatedCharacter(String text) {
92          Character result = null;
93  
94  		if (text == null || text.isEmpty()) {
95  			return null;
96  		}
97  		
98  		result = text.chars() // Stream of String
99  	            .mapToObj(s -> Character.toLowerCase((char) s)) // First convert to Character object and then to lowercase
100 	            .collect(Collectors.groupingBy(Function.identity(), LinkedHashMap::new, Collectors.counting())) //Store the chars in map with count 
101 	            .entrySet()
102 	            .stream()
103 	            .filter(entry -> entry.getValue() == 1L)
104 	            .map(Map.Entry::getKey)
105 	            .findFirst()
106 	            .orElseThrow(() -> new IllegalStateException("No non-repeated character found"));
107 		
108 		return result;
109 	}

140         if (text == null || text.isEmpty()) {
