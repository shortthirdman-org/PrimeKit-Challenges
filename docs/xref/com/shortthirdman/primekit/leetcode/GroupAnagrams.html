10  /**
11   * @author shortthirdman
12   * @apiNote Given an array of strings strs, you need to group all the anagrams together. You can return the answer in any order.
13   */
14  public class GroupAnagrams {
15  
16      /**
17       * <b>Group by Sorting</b><br/>
18       * Time Complexity: O(n * \log k)O(n∗logk) (k is the length of largest string)<br/>
19       * Space Complexity: O(n)O(n)<br/>
20       *
21       * @param contentArr the input text array
22       * @return
23       */
24      public List<List<String>> categorizeBySorting(String[] contentArr) {
25  
26          if (contentArr == null) {
27              throw new NullPointerException("Array input can not empty or null");
28          }
29  
30          if (contentArr.length == 0) {
31              return new ArrayList<>();
32          }
33  
34          Map<String, List<String>> stringAnagramsMap = new HashMap<>();
35          for (String s : contentArr) {
36              char[] arr = s.toCharArray();
37              Arrays.sort(arr);
38              String key = String.valueOf(arr);
39  
40              if (!stringAnagramsMap.containsKey(key)) {
41                  stringAnagramsMap.put(key, new ArrayList<>());
42              }
43  
44              stringAnagramsMap.get(key).add(s);
45          }
46  
47          List<List<String>> resultList = new ArrayList<>();
48          for (Map.Entry<String, List<String>> stringAnagrams : stringAnagramsMap.entrySet()) {
49              resultList.add(stringAnagrams.getValue());
50          }
51          return resultList;
52      }
53  
54      /**
55       * <b>Group by Frequency</b><br/>
56       * Time Complexity: O(n * k)O(n∗k) (k is the length of the largest string)<br/>
57       * Space Complexity: O(n)O(n)<br/>
58       *
59       * @param contentArr the array content
60       * @return
61       */
62      public List<List<String>> categorizeByFrequency(String[] contentArr) {
63  //        if (contentArr == null) {
64  //            throw new NullPointerException("Input array is null");
65  //        }
66  
67          // Check for empty inputs
68          if (contentArr == null || contentArr.length == 0) {
69              return new ArrayList<>();
70          }
71  
72          boolean containsNull = Arrays.stream(contentArr).anyMatch(Objects::isNull);
73          if (containsNull) {
74              throw new IllegalArgumentException("Array contains null values");
75          }
76  
77          Map<String, List<String>> frequencyStringsMap = new HashMap<>();
78          for (String str : contentArr) {
79  
80              String frequencyString = getFrequencyString(str);
81  
82              // If the frequency string is present, add the string to the list
83              if (frequencyStringsMap.containsKey(frequencyString)) {
84                  frequencyStringsMap.get(frequencyString).add(str);
85              } else {
86                  // else create a new list
87                  List<String> strList = new ArrayList<>();
88                  strList.add(str);
89                  frequencyStringsMap.put(frequencyString, strList);
90              }
91          }
92  
93          return new ArrayList<>(frequencyStringsMap.values());
94      }
95  
96      private String getFrequencyString(String element) {
97          // Frequency buckets
98          int[] freq = new int[26];
99  
100         // Iterate over each character
101         for (char c : element.toCharArray()) {
102             freq[c - 'a']++;
103         }
104 
105         // Start creating the frequency string
106         StringBuilder frequencyString = new StringBuilder("");
107         char c = 'a';
108         for (int i : freq) {
109             frequencyString.append(c);
110             frequencyString.append(i);
111             c++;
112         }
113 
114         return frequencyString.toString();
115     }
