3   import com.shortthirdman.primekit.common.Worker;

5   import java.util.*;
6   
7   /**
8    * <b>Minimum Cost to Hire K Workers</b><br/>
9    * There are N workers. The i-th worker has a <code>quality[i]</code> and a minimum wage expectation <code>wage[i]</code>.<br/>
10   *
11   * Now we want to hire exactly K workers to form a paid group. When hiring a group of K workers, we must pay them according to the following rules:<br/>
12   * 1. Every worker in the paid group should be paid in the ratio of their quality compared to other workers in the paid group.<br/>
13   * 2. Every worker in the paid group must be paid at least their minimum wage expectation.<br/>
14   *
15   * Return the least amount of money needed to form a paid group satisfying the above conditions.<br/>
16   *
17   * @author ShortThirdMan
18   */
19  public class HireWorkersMinimumWage {
20  
21  	public double minCostToHireWorkers(int[] quality, int[] wage, int K) {
22          if (quality.length == 0 || wage.length == 0) {
23              throw new IllegalArgumentException("Wages and qualities can not be empty");
24          }
25  
26          if (K > quality.length) {
27              throw new IllegalArgumentException("K cannot be greater than the number of available workers");
28          }
29  
30          if (quality.length != wage.length) {
31              throw new IllegalArgumentException("Mismatched quality and wage for workers");
32          }
33  
34          boolean isNegativeWage = Arrays.stream(wage).boxed().anyMatch(w -> w < 0);
35          boolean isNegativeQuality = Arrays.stream(quality).boxed().anyMatch(q -> q < 0);
36          if (isNegativeQuality || isNegativeWage) {
37              throw new IllegalArgumentException("Wage or quality for worker can not be negative");
38          }
39  
40          int N = quality.length;
41  
42          // Create an array of Worker objects
43          Worker[] workers = new Worker[N];
44          for (int i = 0; i < N; i++) {
45              workers[i] = new Worker(quality[i], wage[i]);
46          }
47  
48          // Sort workers based on the ratio (wage/quality)
49          Arrays.sort(workers, Comparator.comparingDouble(Worker::getRatio));
50  
51          // Priority queue to maintain the largest K qualities
52          PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> b - a);
53          int qualitySum = 0;
54          double minCost = Double.MAX_VALUE;
55  
56          for (Worker worker : workers) {
57              int q = worker.getQuality();
58              qualitySum += q;
59              pq.add(q);
60  
61              // If we exceed the number of workers, remove the smallest quality
62              if (pq.size() > K) {
63                  Integer removedQuality = pq.poll();  // Safely unbox
64                  if (removedQuality != null) {
65                      qualitySum -= removedQuality;
66                  }
67              }
68  
69              // If we have exactly K workers, calculate the cost
70              if (pq.size() == K) {
71                  minCost = Math.min(minCost, worker.getRatio() * qualitySum);
72              }
73          }
74  
75          return minCost;
76      }
77  }
