9   import com.google.gson.Gson;
10  import com.google.gson.GsonBuilder;
11  import com.google.gson.JsonElement;
12  import com.google.gson.JsonObject;

14  public class JsonComparator {
15  	
16  	private final Gson gson;
17  	
18  	public JsonComparator() {
19  		GsonBuilder gsonBuilder = new GsonBuilder();
20  		gsonBuilder.setDateFormat("yyyy-MM-dd HH:mm:ss");
21  		gsonBuilder.serializeNulls();
22  		gsonBuilder.setPrettyPrinting();
23  
24  		// Create and return the custom Gson instance
25  		this.gson = gsonBuilder.create();
26  	}
27  
28  	public List<String> findDifferentKeys(String json1, String json2) {
29  		Set<String> differentKeys = new HashSet<>();
30  		
31  		JsonObject jsonObject1 = gson.fromJson(json1, JsonObject.class);
32          JsonObject jsonObject2 = gson.fromJson(json2, JsonObject.class);
33          
34          for (String key : jsonObject1.keySet()) {
35              if (jsonObject2.has(key)) {
36                  JsonElement value1 = jsonObject1.get(key);
37                  JsonElement value2 = jsonObject2.get(key);
38                  if (!value1.equals(value2)) {
39                      differentKeys.add(key);
40                  }
41              }
42          }
43  
44          List<String> sortedKeys = new ArrayList<>(differentKeys);
45          Collections.sort(sortedKeys);
46          
47          return sortedKeys;
48  	}
49  }
