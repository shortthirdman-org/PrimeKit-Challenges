3   import java.util.Arrays;
4   import java.util.stream.Collectors;
5   
6   public class ReverseWordsGenerator {
7   
8       public String reverseWords(String srcText) {
9           if (srcText == null || srcText.isEmpty()) {
10              throw new UnsupportedOperationException("Input source text can not be blank or null");
11          }
12  
13          String reversedText = Arrays.stream(srcText.split(" "))
14                  .map(word -> new StringBuilder(word).reverse())
15                  .collect(Collectors.joining(" "));
16  
17          String resultText = Arrays.stream(reversedText.split(" "))
18                  .map(word -> Character.toUpperCase(word.charAt(0)) + word.substring(1))
19                  .collect(Collectors.joining(" "));
20  
21          return resultText;
22      }
23  }
