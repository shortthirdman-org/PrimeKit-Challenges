3   import java.util.HashMap;
4   import java.util.Map;
5   
6   public class RomanNumberConverter {
7   
8       private static final Map<Character, Integer> ROMAN_TO_INT = new HashMap<>();
9       private static final Map<Integer, String> INT_TO_ROMAN = new HashMap<>();
10  
11      static {
12          ROMAN_TO_INT.put('I', 1);
13          ROMAN_TO_INT.put('V', 5);
14          ROMAN_TO_INT.put('X', 10);
15          ROMAN_TO_INT.put('L', 50);
16          ROMAN_TO_INT.put('C', 100);
17          ROMAN_TO_INT.put('D', 500);
18          ROMAN_TO_INT.put('M', 1000);
19  
20          INT_TO_ROMAN.put(1, "I");
21          INT_TO_ROMAN.put(4, "IV");
22          INT_TO_ROMAN.put(5, "V");
23          INT_TO_ROMAN.put(9, "IX");
24          INT_TO_ROMAN.put(10, "X");
25          INT_TO_ROMAN.put(40, "XL");
26          INT_TO_ROMAN.put(50, "L");
27          INT_TO_ROMAN.put(90, "XC");
28          INT_TO_ROMAN.put(100, "C");
29          INT_TO_ROMAN.put(400, "CD");
30          INT_TO_ROMAN.put(500, "D");
31          INT_TO_ROMAN.put(900, "CM");
32          INT_TO_ROMAN.put(1000, "M");
33      }
34  
35      public int romanToInteger(String s) {
36          if (s == null) {
37              throw new NullPointerException("Input roman number can not be null");
38          }
39  
40          validateRomanNumeral(s);
41  
42          int result = 0;
43          int prevValue = 0;
44  
45          for (int i = s.length() - 1; i >= 0; i--) {
46              int currentValue = ROMAN_TO_INT.get(s.charAt(i));
47  
48              if (currentValue < prevValue) {
49                  result -= currentValue;
50              } else {
51                  result += currentValue;
52              }
53  
54              prevValue = currentValue;
55          }
56  
57          return result;
58  
59      }
60  
61      public String integerToRoman(int number) {
62          if (number <= 0 || number >= 4000) {
63              throw new IllegalArgumentException("Numbers less than zero (0) or more than 3999 can not be converted to Roman numeral");
64          }
65          StringBuilder result = new StringBuilder();
66  
67          while (number > 0) {
68              int largestDivisor = findLargestDivisor(number);
69              result.append(INT_TO_ROMAN.get(largestDivisor));
70              number -= largestDivisor;
71          }
72  
73          return result.toString();
74      }
75  
76      private int findLargestDivisor(int num) {
77          return INT_TO_ROMAN.keySet().stream()
78                  .filter(divisor -> num >= divisor)
79                  .max(Integer::compare)
80                  .orElse(0);
81      }
82  
83      private void validateRomanNumeral(String s) {
84          // Check if the input contains valid Roman numeral characters
85          for (char c : s.toCharArray()) {
86              if (!ROMAN_TO_INT.containsKey(c)) {
87                  throw new IllegalArgumentException("Invalid Roman numeral character: " + c);
88              }
89          }
90  
91          // Check for invalid repetition patterns
92          if (s.contains("IIII") || s.contains("VV") || s.contains("XXXX") ||
93                  s.contains("LL") || s.contains("CCCC") || s.contains("DD") || s.contains("MMMM")) {
94              throw new IllegalArgumentException("Invalid Roman numeral sequence: " + s);
95          }
96  
97          // Check for invalid subtractive combinations
98          if (s.contains("IL") || s.contains("IC") || s.contains("ID") || s.contains("IM") ||
99                  s.contains("VX") || s.contains("VL") || s.contains("VC") || s.contains("VD") || s.contains("VM") ||
100                 s.contains("XD") || s.contains("XM") ||
101                 s.contains("LC") || s.contains("LD") || s.contains("LM") ||
102                 s.contains("DM")) {
103             throw new IllegalArgumentException("Invalid Roman numeral sequence: " + s);
104         }
105     }
