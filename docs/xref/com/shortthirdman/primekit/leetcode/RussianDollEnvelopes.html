3   import java.util.ArrayList;
4   import java.util.Arrays;
5   import java.util.Comparator;
6   import java.util.List;

8   /**
9    * You have a number of envelopes with widths and heights given as a pair of integers (w, h).
10   * One envelope can fit into another if and only if both the width and height of one envelope is greater than the width and height of the other envelope.
11  
12   * @author shortthirdman
13   * @implNote We can sort the envelopes by height in ascending order and width in descending order. Then look at the width and find the longest increasing subsequence.
14   * @apiNote Time complexity is O(n*log(n)) and we need O(n) of space for the list.
15   * @version 1.0.0
16   */
17  public class RussianDollEnvelopes {
18  
19      public int maxEnvelopes(int[][] envelopes) {
20          if (envelopes == null || envelopes.length == 0) {
21              return 0;
22          }
23  
24          Comparator<int[]> c = Comparator.comparing((int[] arr) -> arr[0])
25                  .thenComparing((int[] arr) -> arr[1], Comparator.reverseOrder());
26          Arrays.sort(envelopes, c);
27  
28          List<Integer> list = new ArrayList<>();
29  
30          for (int[] arr : envelopes) {
31              int target = arr[1];
32  
33              if (list.isEmpty() || target > list.get(list.size() - 1)) {
34                  list.add(target);
35              } else {
36                  int i = 0;
37                  int j = list.size() - 1;
38  
39                  while (i < j) {
40                      int m = i + (j - i) / 2;
41                      if (list.get(m) >= target) {
42                          j = m;
43                      } else {
44                          i = m + 1;
45                      }
46                  }
47  
48                  list.set(j, target);
49              }
50          }
51  
52          return list.size();
53      }
54      
55      public int findMaxEnvelopes(int[][] envelopes) {
56          int n = envelopes.length;
57          // sort by ascending width, and sort by descending height if the width are the same
58          Arrays.sort(envelopes, new Comparator<int[]>() 
59          {
60              public int compare(int[] a, int[] b) {
61                  return a[0] == b[0] ? 
62                      b[1] - a[1] : a[0] - b[0];
63              }
64          });
65          // find LIS on the height array
66          int[] height = new int[n];
67          for (int i = 0; i < n; i++) {
68              height[i] = envelopes[i][1];
69          }
70  
71          return lengthOfLIS(height);
72      }
73      
74      private int lengthOfLIS(int[] nums) {
75          int piles = 0, n = nums.length;
76          int[] top = new int[n];
77          for (int i = 0; i < n; i++) {
78              // playing card to process
79              int poker = nums[i];
80              int left = 0, right = piles;
81              // position to insert for binary search
82              while (left < right) {
83                  int mid = (left + right) / 2;
84                  if (top[mid] >= poker)
85                      right = mid;
86                  else
87                      left = mid + 1;
88              }
89              if (left == piles) piles++;
90              // put this playing cart on top of the pile
91              top[left] = poker;
92          }
93          // the number of cards is the LIS length
94          return piles;
95      }
