3   /**
4    * Given an unsorted array nums, reorder it in-place such that nums[0] &lt; nums[1] &gt; nums[2] &lt; nums[3]....
5    * @author shortthirdman
6    */
7   public class WiggleSort {
8   
9       public void wiggleSort(int[] nums) {
10          if (nums == null || nums.length <= 1) {
11              return;
12          }
13  
14          for (int i = 1; i < nums.length; i++) {
15              if (i % 2 == 1) {
16                  if (nums[i - 1] > nums[i]) {
17                      swap(nums, i - 1, i);
18                  }
19              } else {
20                  if (nums[i - 1] < nums[i]) {
21                      swap(nums, i - 1, i);
22                  }
23              }
24          }
25      }
26  
27      private void swap(int[] nums, int i, int j) {
28          int t = nums[i];
29          nums[i] = nums[j];
30          nums[j] = t;
31      }
32  }
