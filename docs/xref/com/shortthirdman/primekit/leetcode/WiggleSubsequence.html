3   /**
4    * Given a sequence of integers, return the length of the longest subsequence that is a wiggle sequence.
5    * A subsequence is obtained by deleting some number of elements (eventually, also zero) from the original sequence,
6    * leaving the remaining elements in their original order.
7    *
8    * @author shortthirdman
9    */
10  public class WiggleSubsequence {
11  
12      public int wiggleMaxLength(int[] nums) {
13          if (nums == null || nums.length == 0) {
14              return 0;
15          }
16  
17          if (nums.length < 2) {
18              return nums.length;
19          }
20  
21          int count = 1;
22  
23  
24          for (int i = 1, j = 0; i < nums.length; j = i, i++) {
25              if (nums[j] < nums[i]) {
26                  count++;
27                  while (i < nums.length - 1 && nums[i] <= nums[i + 1]) {
28                      i++;
29                  }
30              } else if (nums[j] > nums[i]) {
31                  count++;
32                  while (i < nums.length - 1 && nums[i] >= nums[i + 1]) {
33                      i++;
34                  }
35              }
36          }
37  
38          return count;
39      }
40  }
