3   import org.apache.commons.lang3.StringUtils;
4   
5   import java.util.*;
6   
7   /**
8    * Given a string s and a dictionary of words dict, add spaces in s to construct a sentence where each word is a valid dictionary word.
9    * Return all such possible sentences.
10   */
11  public class WordBreak {
12  
13  
14      @SuppressWarnings({"rawtypes", "unchecked"})
15      public List<String> wordBreak(String s, Set<String> dict) {
16          if (Objects.isNull(s)) {
17              throw new NullPointerException("Input string can not be null");
18          }
19  
20          if (StringUtils.isBlank(s)) {
21              return List.of();
22          }
23  
24          ArrayList[] dp = new ArrayList[s.length() + 1];
25          dp[0] = new ArrayList<>();
26  
27          for (int i = 0; i < s.length(); i++) {
28              if (dp[i] == null)
29                  continue;
30  
31              for (String word : dict) {
32                  int len = word.length();
33                  int end = i + len;
34                  if (end > s.length())
35                      continue;
36  
37                  if (s.substring(i, end).equals(word)) {
38                      if (dp[end] == null) {
39                          dp[end] = new ArrayList<>();
40                      }
41                      dp[end].add(word);
42                  }
43              }
44          }
45  
46          List<String> result = new LinkedList<>();
47          if (dp[s.length()] == null) {
48              return result;
49          }
50  
51          ArrayList<String> temp = new ArrayList<>();
52          dfs(dp, s.length(), result, temp);
53  
54          return result;
55      }
56  
57      private void dfs(List<String>[] dp, int end, List<String> result, ArrayList<String> tmp) {
58          if (end <= 0) {
59              StringBuilder path = new StringBuilder(tmp.get(tmp.size() - 1));
60              for (int i = tmp.size() - 2; i >= 0; i--) {
61                  path.append(" ").append(tmp.get(i));
62              }
63  
64              result.add(path.toString());
65              return;
66          }
67  
68          for (String str : dp[end]) {
69              tmp.add(str);
70              dfs(dp, end - str.length(), result, tmp);
71              tmp.remove(tmp.size() - 1);
72          }
73      }
74  }
